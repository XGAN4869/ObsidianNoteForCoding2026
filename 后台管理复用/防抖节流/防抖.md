```js
//debounce 第一版, 无 args  
export function debounce1(fn,delay){  
  let timer;  
  return function(value){  
    clearTimeout(timer);  
    timer = setTimeout(() => fn(value), delay);  
  }  
}  
  
//debounce 第二版,无this  
export function debounce2(fn, delay) {  
  let timer  
  return function (...args) {  
    clearTimeout(timer)  
    timer = setTimeout(() => fn(...args), delay)  
  }  
}  
  
//第三版都有，无 args 会导致参数传不全，无 this 导致内部有 this 相关的函数无法正常输出
export function debounce(fn, delay) {  
  let timer;  
  // ...args 会收集所有传入的参数，变成一个数组  
  return function (...args) {  
    clearTimeout(timer);  
    timer = setTimeout(() => fn.apply(this, args), delay);  
  };  
}
```
