### Proto 定义
```protobuf
service AuthService {
  rpc Register(RegisterRequest) returns (RegisterResponse);
  rpc Login(LoginRequest) returns (LoginResponse);
  rpc ValidateToken(ValidateTokenRequest) returns (ValidateTokenResponse);
  rpc GetProfile(GetProfileRequest) returns (User);
}
```

### 核心代码
```go
func (s *authServer) Login(ctx context.Context, req *pb.LoginRequest) (*pb.LoginResponse, error) {
	// 验证用户名密码 → 生成 JWT
	token, _ := generateJWT(user.ID, user.Username)
	return &pb.LoginResponse{Token: token, ExpiresIn: 86400}, nil
}

func (s *authServer) ValidateToken(ctx context.Context, req *pb.ValidateTokenRequest) (*pb.ValidateTokenResponse, error) {
	claims, err := parseJWT(req.GetToken())
	if err != nil {
		return nil, status.Error(codes.Unauthenticated, "Token无效")
	}
	return &pb.ValidateTokenResponse{UserId: claims.UserID, Valid: true}, nil
}
```

### API
| RPC | 说明 |
|-----|------|
| Register | 注册 |
| Login | 登录 |
| ValidateToken | 验证 Token |
| GetProfile | 获取用户信息 |
