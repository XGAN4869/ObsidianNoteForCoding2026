## 二、localStorage 工具

新建 `src/lib/todos.ts`：

```ts
import type { Todo, TodoFilter } from '@/types/todo';

const STORAGE_KEY = 'pixel-blog-todos';

export function getAllTodosClient(): Todo[] {
  if (typeof window === 'undefined') return [];
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
}

export function saveAllTodosClient(todos: Todo[]): void {
  if (typeof window === 'undefined') return;
  localStorage.setItem(STORAGE_KEY, JSON.stringify(todos));
}

export function addTodoClient(input: Omit<Todo, 'id' | 'createdAt' | 'done'>): Todo {
  const todos = getAllTodosClient();
  const newTodo: Todo = {
    id: cryptoRandomUUID(),
    createdAt: new Date().toISOString(),
    done: false,
    ...input,
  };
  todos.push(newTodo);
  saveAllTodosClient(todos);
  return newTodo;
}

export function toggleTodoClient(id: string): void {
  const todos = getAllTodosClient();
  const todo = todos.find((t) => t.id === id);
  if (todo) {
    todo.done = !todo.done;
    saveAllTodosClient(todos);
  }
}

export function updateTodoClient(id: string, patch: Partial<Todo>): void {
  const todos = getAllTodosClient();
  const todo = todos.find((t) => t.id === id);
  if (todo) {
    Object.assign(todo, patch);
    saveAllTodosClient(todos);
  }
}

export function deleteTodoClient(id: string): void {
  const todos = getAllTodosClient();
  saveAllTodosClient(todos.filter((t) => t.id !== id));
}

export function clearCompletedTodosClient(): void {
  const todos = getAllTodosClient();
  saveAllTodosClient(todos.filter((t) => !t.done));
}

export function filterTodos(todos: Todo[], filter: TodoFilter): Todo[] {
  switch (filter) {
    case 'active':
      return todos.filter((t) => !t.done);
    case 'completed':
      return todos.filter((t) => t.done);
    default:
      return todos;
  }
}

export function sortTodos(todos: Todo[]): Todo[] {
  return [...todos].sort((a, b) => {
    // 未完成的在前
    if (a.done !== b.done) return a.done ? 1 : -1;
    // 高优先级在前
    const priorityOrder = { high: 0, normal: 1, low: 2 };
    const pa = priorityOrder[a.priority ?? 'normal'];
    const pb = priorityOrder[b.priority ?? 'normal'];
    if (pa !== pb) return pa - pb;
    // 截止日期近的在前
    if (a.dueDate && b.dueDate) {
      return a.dueDate.localeCompare(b.dueDate);
    }
    if (a.dueDate) return -1;
    if (b.dueDate) return 1;
    // 创建早的在前
    return a.createdAt.localeCompare(b.createdAt);
  });
}

function cryptoRandomUUID(): string {
  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
    return crypto.randomUUID();
  }
  return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, (c) => {
    const r = (Math.random() * 16) | 0;
    const v = c === 'x' ? r : (r & 0x3) | 0x8;
    return v.toString(16);
  });
}
```

## 三、代办清单页

新建 `src/app/todos/page.tsx`：

```tsx
'use client';

import { useEffect, useState } from 'react';
import {
  getAllTodosClient,
  addTodoClient,
  toggleTodoClient,
  deleteTodoClient,
  clearCompletedTodosClient,
  filterTodos,
  sortTodos,
} from '@/lib/todos';
import { TodoItem } from '@/components/todo/TodoItem';
import { TodoForm } from '@/components/todo/TodoForm';
import { TodoFilters } from '@/components/todo/TodoFilters';
import { PixelCard, PixelButton, PixelEmpty } from '@/components/pixel';
import { Container } from '@/components/layout/Container';
import type { Todo, TodoFilter } from '@/types/todo';

export default function TodosPage() {
  const [todos, setTodos] = useState<Todo[]>([]);
  const [filter, setFilter] = useState<TodoFilter>('all');
  const [loaded, setLoaded] = useState(false);

  useEffect(() => {
    setTodos(getAllTodosClient());
    setLoaded(true);
  }, []);

  const handleAdd = (input: { title: string; dueDate?: string; priority?: 'low' | 'normal' | 'high' }) => {
    const newTodo = addTodoClient(input);
    setTodos((prev) => [...prev, newTodo]);
  };

  const handleToggle = (id: string) => {
    toggleTodoClient(id);
    setTodos((prev) =>
      prev.map((t) => (t.id === id ? { ...t, done: !t.done } : t)),
    );
  };

  const handleDelete = (id: string) => {
    deleteTodoClient(id);
    setTodos((prev) => prev.filter((t) => t.id !== id));
  };

  const handleClearCompleted = () => {
    if (!confirm('确认清除所有已完成？')) return;
    clearCompletedTodosClient();
    setTodos((prev) => prev.filter((t) => !t.done));
  };

  const sorted = sortTodos(todos);
  const filtered = filterTodos(sorted, filter);

  const stats = {
    total: todos.length,
    active: todos.filter((t) => !t.done).length,
    completed: todos.filter((t) => t.done).length,
  };

  return (
    <Container>
      <header className="flex items-center justify-between mb-6 flex-wrap gap-4">
        <h1 className="text-4xl font-bold">[ 代办清单 ]</h1>
        {stats.completed > 0 && (
          <PixelButton onClick={handleClearCompleted}>
            清除已完成 ({stats.completed})
          </PixelButton>
        )}
      </header>

      {/* 统计 */}
      <PixelCard className="mb-6">
        <div className="p-4 flex items-center justify-around gap-4 flex-wrap text-center">
          <Stat label="全部" value={stats.total} />
          <span className="text-2xl opacity-50">|</span>
          <Stat label="待完成" value={stats.active} highlight />
          <span className="text-2xl opacity-50">|</span>
          <Stat label="已完成" value={stats.completed} />
        </div>
      </PixelCard>

      {/* 新增表单 */}
      <TodoForm onAdd={handleAdd} />

      {/* 过滤器 */}
      {todos.length > 0 && (
        <TodoFilters value={filter} onChange={setFilter} counts={stats} />
      )}

      {/* 列表 */}
      {!loaded ? (
        <p className="mt-4">加载中...</p>
      ) : filtered.length === 0 ? (
        <PixelEmpty
          icon="[·]"
          title={filter === 'all' ? '还没有代办' : '这里空空如也'}
          description={filter === 'all' ? '在上方表单里加一个吧' : '切换到"全部"看看别的'}
        />
      ) : (
        <ul className="mt-4 flex flex-col gap-2">
          {filtered.map((todo) => (
            <TodoItem
              key={todo.id}
              todo={todo}
              onToggle={handleToggle}
              onDelete={handleDelete}
            />
          ))}
        </ul>
      )}
    </Container>
  );
}

function Stat({ label, value, highlight }: { label: string; value: number; highlight?: boolean }) {
  return (
    <div className="flex flex-col gap-1">
      <span className="text-3xl font-bold">{value}</span>
      <span className="text-sm">{label}</span>
    </div>
  );
}
```

'use client';

import { useState } from 'react';
import {
  PixelCard,
  PixelInput,
  PixelSelect,
  PixelButton,
} from '@/components/pixel';

type Props = {
  onAdd: (input: {
    title: string;
    dueDate?: string;
    priority?: 'low' | 'normal' | 'high';
  }) => void;
};

export function TodoForm({ onAdd }: Props) {
  const [title, setTitle] = useState('');
  const [dueDate, setDueDate] = useState('');
  const [priority, setPriority] = useState<'low' | 'normal' | 'high'>('normal');

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!title.trim()) return;

    onAdd({
      title: title.trim(),
      dueDate: dueDate || undefined,
      priority,
    });

    setTitle('');
    setDueDate('');
    setPriority('normal');
  };

  return (
    <PixelCard>
      <form onSubmit={handleSubmit} className="p-4 flex flex-col gap-3">
        <PixelInput
          label="任务"
          placeholder="今天要做啥..."
          value={title}
          onChange={(e) => setTitle(e.target.value)}
          maxLength={200}
          required
        />
        <div className="grid grid-cols-2 gap-3">
          <PixelInput
            label="截止日期（可选）"
            type="date"
            value={dueDate}
            onChange={(e) => setDueDate(e.target.value)}

新建 `src/components/todo/TodoItem.tsx`：

```tsx
'use client';

import { PixelCard } from '@/components/pixel';
import { formatDate } from '@/lib/date';
import type { Todo } from '@/types/todo';

type Props = {
  todo: Todo;
  onToggle: (id: string) => void;
  onDelete: (id: string) => void;
};

const priorityIcons = {
  low: '·',
  normal: '·',
  high: '!',
};

const priorityLabels = {
  low: '低',
  normal: '中',
  high: '高',
};

export function TodoItem({ todo, onToggle, onDelete }: Props) {
  const isOverdue =
    todo.dueDate && !todo.done && new Date(todo.dueDate) < new Date();

  return (
    <li>
      <PixelCard shadow={todo.done ? 'none' : 'sm'}>
        <div
          className={`
            p-3 flex items-start gap-3
            ${todo.done ? 'opacity-50' : ''}
          `}
        >
          {/* 自定义勾选框 */}
          <button
            onClick={() => onToggle(todo.id)}
            className="
              w-6 h-6 mt-0.5 shrink-0
              border-2 border-fg bg-bg
              flex items-center justify-center
              cursor-pointer
              font-bold
            "
            aria-label={todo.done ? '标记为未完成' : '标记为完成'}
          >
            {todo.done ? '✓' : ''}
          </button>

          <div className="flex-1 min-w-0">
            <p
              className={`
                break-words
                ${todo.done ? 'line-through' : ''}
              `}
            >
              {todo.priority === 'high' && !todo.done && (
                <span className="mr-1">[!]</span>
              )}
              {todo.title}
            </p>

            {(todo.dueDate || todo.priority) && (
              <p className="text-xs mt-1 opacity-70">
                {todo.dueDate && (
                  <span className={isOverdue ? 'font-bold' : ''}>
                    [{formatDate(todo.dueDate)}]{isOverdue ? ' [已过期]' : ''}
                  </span>
                )}
                {todo.priority && (
                  <span className="ml-2">[{priorityLabels[todo.priority!]}]</span>
                )}
              </p>
            )}
          </div>

          <button
            onClick={() => {
              if (confirm('删除这个任务？')) onDelete(todo.id);
            }}
            className="
              shrink-0
              border-2 border-fg
              px-2 py-0.5
              hover:bg-fg hover:text-bg
              transition-none
            "
            aria-label="删除"
          >
            [X]
          </button>
        </div>
      </PixelCard>
    </li>
  );
}
```

import { PixelButton } from '@/components/pixel';
import type { TodoFilter } from '@/types/todo';

          key={f.value}
          variant={value === f.value ? 'primary' : 'default'}
          onClick={() => onChange(f.value)}
        >
          {f.label} ({counts[f.value]})
        </PixelButton>
      ))}
    </div>
  );
}

### 改造 todos 页加 DndContext

```tsx
import { DndContext, closestCenter, KeyboardSensor, PointerSensor, useSensor, useSensors, type DragEndEvent } from '@dnd-kit/core';
import { SortableContext, sortableKeyboardCoordinates, verticalListSortingStrategy, arrayMove } from '@dnd-kit/sortable';

// 在 return 外面加：
const sensors = useSensors(
  useSensor(PointerSensor),
  useSensor(KeyboardSensor, { coordinateGetter: sortableKeyboardCoordinates }),
);

const handleDragEnd = (event: DragEndEvent) => {
  const { active, over } = event;
  if (over && active.id !== over.id) {
    setTodos((prev) => {
      const oldIndex = prev.findIndex((t) => t.id === active.id);
      const newIndex = prev.findIndex((t) => t.id === over.id);
      const newTodos = arrayMove(prev, oldIndex, newIndex);
      saveAllTodosClient(newTodos);
      return newTodos;
    });
  }
};

// return 里把 <ul> 改成：
<DndContext sensors={sensors} collisionDetection={closestCenter} onDragEnd={handleDragEnd}>
  <SortableContext items={filtered.map((t) => t.id)} strategy={verticalListSortingStrategy}>
    <ul className="mt-4 flex flex-col gap-2">
      {filtered.map((todo) => (
        <TodoItem
          key={todo.id}
          todo={todo}
          onToggle={handleToggle}
          onDelete={handleDelete}
        />
      ))}
    </ul>
  </SortableContext>
</DndContext>
```
