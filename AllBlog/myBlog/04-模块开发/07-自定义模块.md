### 音乐播放器组件

新建 `src/components/music/MusicPlayer.tsx`：

```tsx
'use client';

import { useEffect, useRef, useState } from 'react';
import type { Music } from '@/types/music';

type Props = {
  music: Music;
  onNext?: () => void;
  onPrev?: () => void;
};

export function MusicPlayer({ music, onNext, onPrev }: Props) {
  const audioRef = useRef<HTMLAudioElement>(null);
  const [playing, setPlaying] = useState(false);
  const [progress, setProgress] = useState(0);
  const [duration, setDuration] = useState(0);

  useEffect(() => {
    // 切换歌曲时自动播放
    audioRef.current?.play().catch(() => {});
  }, [music.url]);

  const togglePlay = () => {
    if (!audioRef.current) return;
    if (playing) {
      audioRef.current.pause();
    } else {
      audioRef.current.play().catch(() => {});
    }
    setPlaying(!playing);
  };

  const handleTimeUpdate = () => {
    if (!audioRef.current) return;
    const current = audioRef.current.currentTime;
    setProgress(current);
    setDuration(audioRef.current.duration || 0);
  };

  return (
    <div className="border-2 border-fg p-4 flex flex-col gap-3">
      <audio
        ref={audioRef}
        src={music.url}
        onTimeUpdate={handleTimeUpdate}
        onEnded={() => setPlaying(false)}
      />

      {/* 歌曲信息 */}
      <div className="flex items-center gap-3">
        {music.cover ? (
          <img
            src={music.cover}
            alt={music.title}
            className="w-16 h-16 border-2 border-fg object-cover"
          />
        ) : (
          <div className="w-16 h-16 border-2 border-fg flex items-center justify-center text-2xl">
            ♪
          </div>
        )}
        <div className="flex-1 min-w-0">
          <h3 className="text-xl font-bold truncate">{music.title}</h3>
          <p className="text-sm truncate">{music.artist}</p>
        </div>
      </div>

      {/* 进度条 */}
      <div className="flex items-center gap-2 text-xs">
        <span>{formatTime(progress)}</span>
        <div className="flex-1 h-4 border-2 border-fg relative">
          <div
            className="absolute inset-y-0 left-0 bg-fg"
            style={{ width: `${duration ? (progress / duration) * 100 : 0}%` }}
          />
        </div>
        <span>{formatTime(duration)}</span>
      </div>

      {/* 控制按钮 */}
      <div className="flex items-center justify-center gap-2">
        {onPrev && (
          <button
            onClick={onPrev}
            className="border-2 border-fg px-3 py-1 hover:bg-fg hover:text-bg"
            aria-label="上一首"
          >
            [⏮]
          </button>
        )}
        <button
          onClick={togglePlay}
          className="border-2 border-fg px-4 py-2 bg-fg text-bg"
          aria-label={playing ? '暂停' : '播放'}
        >
          {playing ? '[⏸]' : '[▶]'}
        </button>
        {onNext && (
          <button
            onClick={onNext}
            className="border-2 border-fg px-3 py-1 hover:bg-fg hover:text-bg"
            aria-label="下一首"
          >
            [⏭]
          </button>
        )}
      </div>
    </div>
  );
}

function formatTime(seconds: number): string {
  if (!seconds || !isFinite(seconds)) return '0:00';
  const m = Math.floor(seconds / 60);
  const s = Math.floor(seconds % 60);
  return `${m}:${s.toString().padStart(2, '0')}`;
}
```

'use client';

import { useState } from 'react';
import { getAllMusic } from '@/lib/music';
import { MusicPlayer } from '@/components/music/MusicPlayer';
import { PixelCard, PixelEmpty } from '@/components/pixel';
import { Container } from '@/components/layout/Container';

export default function MusicPage() {
  const music = getAllMusic();
  const [currentIdx, setCurrentIdx] = useState(0);

  if (music.length === 0) {
    return (

        music={current}
        onPrev={() => setCurrentIdx((i) => (i - 1 + music.length) % music.length)}
        onNext={() => setCurrentIdx((i) => (i + 1) % music.length)}
      />

      {/* 歌单 */}
      <PixelCard className="mt-6">
        <h2 className="text-2xl font-bold p-4 border-b-2 border-fg">歌单 ({music.length})</h2>
        <ul className="p-4 flex flex-col gap-2">
          {music.map((m, i) => (
            <li key={m.id}>
              <button
                onClick={() => setCurrentIdx(i)}
                className={`
                  w-full text-left border-2 border-fg p-2
                  ${i === currentIdx ? 'bg-fg text-bg' : 'hover:bg-fg hover:text-bg'}
                `}
              >
                {m.title} - {m.artist}
              </button>
            </li>
          ))}
        </ul>
      </PixelCard>
    </Container>
  );
}

## 五、📷 像素相册

> 上传图片 → 自动转 1-bit 像素风 → 显示在博客。

### 实现原理

1. 用户上传图片 → 存到 `public/gallery/`
2. 用 Canvas 把图片缩到 128x128（或 64x64）
3. 把每个像素**和阈值比较**，变成纯黑或纯白
4. 再用 `image-rendering: pixelated` 放大显示

### 安装

```bash
pnpm add -D @types/node
```

### 工具函数

新建 `src/lib/pixelate.ts`：

```ts
/**
 * 把图片转成 1-bit 像素风
 * @param imageUrl 原图 URL
 * @param targetSize 目标像素尺寸（默认 64）
 * @param threshold 黑白阈值 0-255（默认 128）
 * @returns Promise<HTMLCanvasElement>
 */
export async function pixelate(
  imageUrl: string,
  targetSize = 64,
  threshold = 128,
): Promise<HTMLCanvasElement> {
  const img = new Image();
  img.crossOrigin = 'anonymous';

  await new Promise((resolve, reject) => {
    img.onload = resolve;
    img.onerror = reject;
    img.src = imageUrl;
  });

  const canvas = document.createElement('canvas');
  canvas.width = targetSize;
  canvas.height = targetSize;

  const ctx = canvas.getContext('2d');
  if (!ctx) throw new Error('Canvas not supported');

  // 先缩到目标大小
  ctx.imageSmoothingEnabled = true;
  ctx.drawImage(img, 0, 0, targetSize, targetSize);

  // 取每个像素，转 1-bit
  const imageData = ctx.getImageData(0, 0, targetSize, targetSize);
  const data = imageData.data;

  for (let i = 0; i < data.length; i += 4) {
    // 取 RGB 平均亮度
    const brightness = (data[i] + data[i + 1] + data[i + 2]) / 3;
    const value = brightness > threshold ? 255 : 0;
    data[i] = value;     // R
    data[i + 1] = value; // G
    data[i + 2] = value; // B
    // data[i + 3] 是 alpha，不动
  }

  ctx.putImageData(imageData, 0, 0);
  return canvas;
}
```

### 相册页

新建 `src/app/custom/gallery/page.tsx`：

```tsx
'use client';

import { useEffect, useState } from 'react';
import galleryData from '@/../data/gallery.json';
import { pixelate } from '@/lib/pixelate';
import { Container } from '@/components/layout/Container';

type GalleryItem = { id: string; url: string; caption: string };

export default function GalleryPage() {
  const items = galleryData as GalleryItem[];
  const [pixelated, setPixelated] = useState<Map<string, string>>(new Map());

  useEffect(() => {
    void (async () => {
      const map = new Map<string, string>();
      for (const item of items) {
        try {
          const canvas = await pixelate(item.url, 64);
          map.set(item.id, canvas.toDataURL('image/png'));
        } catch (e) {
          console.error('像素化失败:', item.url, e);
        }
      }
      setPixelated(map);
    })();
  }, [items]);

  return (
    <Container>
      <h1 className="text-4xl font-bold mb-6">[ 像素相册 ]</h1>

      <div className="grid grid-cols-2 sm:grid-cols-3 gap-4">
        {items.map((item) => (
          <figure key={item.id} className="border-2 border-fg p-2">
            {pixelated.has(item.id) ? (
              <img
                src={pixelated.get(item.id)}
                alt={item.caption}
                className="w-full h-auto"
                style={{ imageRendering: 'pixelated' }}
              />
            ) : (
              <div className="w-full aspect-square border border-fg flex items-center justify-center">
                转换中...
              </div>
            )}
            <figcaption className="text-sm mt-2 text-center">
              {item.caption}
            </figcaption>
          </figure>
        ))}
      </div>
    </Container>
  );
}
```

> ⚠️ **性能问题**：每张图都用浏览器像素化，**大相册很慢**。
> 真实项目应该用 **服务端预渲染**（ImageMagick 或 Sharp）。
> 个人博客量小，**浏览器内转就够**。

## 六、🌧 心情日历

> 每天选一个像素表情，一年下来能看心情变化。

### 数据结构

```ts
export type MoodEntry = {
  date: string;        // 'YYYY-MM-DD'
  mood: 'happy' | 'calm' | 'sad' | 'angry' | 'tired';
};
```

存储：用 `localStorage`（每用户自己的），或者后端（共享）。

### 心情映射

```ts
const MOOD_ICONS = {
  happy: '☼',
  calm: '·',
  sad: '☁',
  angry: '!',
  tired: 'z',
};
```

### 页面

新建 `src/app/custom/mood/page.tsx`：

```tsx
'use client';

import { useEffect, useState } from 'react';
import { Container } from '@/components/layout/Container';
import { PixelCard, PixelButton } from '@/components/pixel';

const MOODS = [
  { key: 'happy', icon: '☼', label: '开心' },
  { key: 'calm', icon: '·', label: '平静' },
  { key: 'sad', icon: '☁', label: '难过' },
  { key: 'angry', icon: '!', label: '生气' },
  { key: 'tired', icon: 'z', label: '累' },
] as const;

type MoodKey = typeof MOODS[number]['key'];

const STORAGE_KEY = 'pixel-blog-mood';

type MoodMap = Record<string, MoodKey>;

export default function MoodPage() {
  const [moods, setMoods] = useState<MoodMap>({});

  useEffect(() => {
    const stored = localStorage.getItem(STORAGE_KEY);
    if (stored) setMoods(JSON.parse(stored));
  }, []);

  const today = new Date().toISOString().slice(0, 10);

  const setTodayMood = (mood: MoodKey) => {
    const next = { ...moods, [today]: mood };
    setMoods(next);
    localStorage.setItem(STORAGE_KEY, JSON.stringify(next));
  };

  // 生成过去 365 天的网格
  const days = Array.from({ length: 365 }, (_, i) => {
    const d = new Date();
    d.setDate(d.getDate() - i);
    return d.toISOString().slice(0, 10);
  });

  return (
    <Container>
      <h1 className="text-4xl font-bold mb-6">[ 心情日历 ]</h1>

      {/* 今日选 */}
      <PixelCard className="mb-6">
        <div className="p-4 flex flex-col gap-3">
          <h2 className="text-xl font-bold">今天的心情？</h2>
          <div className="flex gap-2 flex-wrap">
            {MOODS.map((m) => (
              <PixelButton
                key={m.key}
                variant={moods[today] === m.key ? 'primary' : 'default'}
                onClick={() => setTodayMood(m.key)}
              >
                [{m.icon}] {m.label}
              </PixelButton>
            ))}
          </div>
        </div>
      </PixelCard>

      {/* 历史 */}
      <PixelCard>
        <h2 className="text-2xl font-bold p-4 border-b-2 border-fg">过去 365 天</h2>
        <div className="p-4 grid grid-cols-15 gap-px">
          {days.map((date) => {
            const mood = moods[date];
            const moodIcon = mood ? MOODS.find((m) => m.key === mood)?.icon : '·';
            return (
              <div
                key={date}
                className="aspect-square border border-fg flex items-center justify-center text-sm"
                title={date}
              >
                {moodIcon}
              </div>
            );
          })}
        </div>
        <div className="p-4 text-sm flex gap-3 flex-wrap">
          {MOODS.map((m) => (
            <span key={m.key}>[{m.icon}] {m.label}</span>
          ))}
        </div>
      </PixelCard>
    </Container>
  );
}
```

> 365 天 × 每格 = **365 个 div**，浏览器能扛。
