# 02 · 时光胶囊

> 给未来的自己写信。
> 不到时间，信封是锁住的，看不到内容。

---

## 一、什么是"时光胶囊"

```
┌──────────────────────────┐
│      🔒 信封（已锁）       │
│                          │
│   寄给未来的自己          │
│   解锁时间：2027-01-01    │
│   还有 120 天            │
└──────────────────────────┘

┌──────────────────────────┐
│      💌 信封（已解锁）     │
│                          │
│   2027 年 1 月 1 日，     │
│   你好，3 年前的我。     │
│                          │
│   [胶囊内容...]          │
└──────────────────────────┘
```

**逻辑**：

- 现在 < `unlockAt`：显示 🔒，倒计时
- 现在 >= `unlockAt`：显示完整内容

**注意**：因为是**纯前端判断**，**用户改本地时间就能看到未解锁内容**。

> 这是个**纯娱乐模块**，不是银行密码箱。
> 别用来存真密码、真隐私。

---

## 二、存储方案选择

| 方案 | 优点 | 缺点 |
| --- | --- | --- |
| **A. localStorage（推荐）** | 零依赖、不需要后端、即时 | 换设备丢数据 |
| **B. JSON + GitHub API** | 跨设备 | 写有 1-2 秒延迟 |

**默认用 A**（简单）。如果你想跨设备，看 `04-进阶`。

## 三、数据结构

新建 `src/types/capsule.ts`：

```ts
export type Capsule = {
  id: string;            // UUID
  title: string;         // 胶囊标题
  content: string;       // 胶囊内容
  unlockAt: string;      // ISO 8601 datetime: '2027-01-01T00:00:00.000Z'
  createdAt: string;     // ISO 8601 datetime
};

export type CapsuleWithStatus = Capsule & {
  unlocked: boolean;     // 是否已解锁
  daysUntilUnlock: number; // 距离解锁还有几天（负数表示已过期 N 天）
};
```

新建 `src/lib/capsules.ts`：

```ts
import type { Capsule, CapsuleWithStatus } from '@/types/capsule';

/**
 * 客户端：localStorage 读写
 * 服务端：用不到
 */
const STORAGE_KEY = 'pixel-blog-capsules';

/**
 * 读取所有胶囊（仅客户端）
 */
export function getAllCapsulesClient(): Capsule[] {
  if (typeof window === 'undefined') return [];
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    return raw ? JSON.parse(raw) : [];
  } catch {
    return [];
  }
}

/**
 * 写入胶囊（仅客户端）
 */
export function saveAllCapsulesClient(capsules: Capsule[]): void {
  if (typeof window === 'undefined') return;
  localStorage.setItem(STORAGE_KEY, JSON.stringify(capsules));
}

/**
 * 加一条胶囊
 */
export function addCapsuleClient(input: Omit<Capsule, 'id' | 'createdAt'>): Capsule {
  const capsules = getAllCapsulesClient();
  const newCapsule: Capsule = {
    id: cryptoRandomUUID(),
    createdAt: new Date().toISOString(),
    ...input,
  };
  capsules.push(newCapsule);
  saveAllCapsulesClient(capsules);
  return newCapsule;
}

/**
 * 删一条
 */
export function deleteCapsuleClient(id: string): void {
  const capsules = getAllCapsulesClient();
  const filtered = capsules.filter((c) => c.id !== id);
  saveAllCapsulesClient(filtered);
}

/**
 * 计算胶囊状态
 */
export function withStatus(capsule: Capsule, now: Date = new Date()): CapsuleWithStatus {
  const unlockAt = new Date(capsule.unlockAt);
  const diffMs = unlockAt.getTime() - now.getTime();
  const days = Math.ceil(diffMs / (24 * 60 * 60 * 1000));
  return {
    ...capsule,
    unlocked: diffMs <= 0,
    daysUntilUnlock: days,
  };
}

/**
 * UUID 生成
 */
function cryptoRandomUUID(): string {
  if (typeof crypto !== 'undefined' && crypto.randomUUID) {
    return crypto.randomUUID();
  }
  // 降级方案（老浏览器）
  return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, (c) => {
    const r = (Math.random() * 16) | 0;
    const v = c === 'x' ? r : (r & 0x3) | 0x8;
    return v.toString(16);
  });
}
```

## 四、列表页

新建 `src/app/capsules/page.tsx`：

```tsx
'use client';

import { useEffect, useState } from 'react';
import Link from 'next/link';
import {
  getAllCapsulesClient,
  withStatus,
} from '@/lib/capsules';
import { PixelCard, PixelButton, PixelEmpty } from '@/components/pixel';
import { Container } from '@/components/layout/Container';
import { formatDate } from '@/lib/date';
import type { CapsuleWithStatus } from '@/types/capsule';

export default function CapsulesPage() {
  const [capsules, setCapsules] = useState<CapsuleWithStatus[]>([]);
  const [loaded, setLoaded] = useState(false);

  useEffect(() => {
    const stored = getAllCapsulesClient();
    setCapsules(stored.map((c) => withStatus(c)));
    setLoaded(true);

    // 每分钟刷新倒计时
    const timer = setInterval(() => {
      setCapsules((prev) =>
        prev.map((c) => withStatus({ ...c })),
      );
    }, 60_000);

    return () => clearInterval(timer);
  }, []);

  const sorted = [...capsules].sort((a, b) =>
    a.unlockAt.localeCompare(b.unlockAt),
  );

  return (
    <Container>
      <header className="flex items-center justify-between mb-6 flex-wrap gap-4">
        <h1 className="text-4xl font-bold">[ 时光胶囊 ]</h1>
        <Link href="/capsules/new">
          <PixelButton variant="primary">[+ 写新胶囊]</PixelButton>
        </Link>
      </header>

      {!loaded ? (
        <p>加载中...</p>
      ) : sorted.length === 0 ? (
        <PixelEmpty
          icon="[💌]"
          title="还没有时光胶囊"
          description="给未来的自己写一封信吧"
          action={
            <Link href="/capsules/new">
              <PixelButton variant="primary">写第一封</PixelButton>
            </Link>
          }
        />
      ) : (
        <ul className="flex flex-col gap-4">
          {sorted.map((capsule) => (
            <li key={capsule.id}>
              <CapsuleCard capsule={capsule} />
            </li>
          ))}
        </ul>
      )}
    </Container>
  );
}

function CapsuleCard({ capsule }: { capsule: CapsuleWithStatus }) {
  return (
    <PixelCard>
      <div className="p-4 flex flex-col gap-2">
        <div className="flex items-start justify-between gap-4">
          <h2 className="text-2xl font-bold">{capsule.title}</h2>
          <span className="text-2xl">
            {capsule.unlocked ? '[💌]' : '[🔒]'}
          </span>
        </div>
        <p className="text-sm">
          解锁时间：{formatDate(capsule.unlockAt)}
        </p>
        {capsule.unlocked ? (
          <div className="mt-2 border-t-2 border-fg pt-3">
            <p>{capsule.content}</p>
          </div>
        ) : (
          <p className="text-sm">
            {capsule.daysUntilUnlock > 0
              ? `还有 ${capsule.daysUntilUnlock} 天解锁`
              : '即将解锁'}
          </p>
        )}
        <p className="text-xs mt-2 opacity-70">
          创建于 {formatDate(capsule.createdAt)}
        </p>
      </div>
    </PixelCard>
  );
}
```

## 五、新增胶囊页

新建 `src/app/capsules/new/page.tsx`：

```tsx
'use client';

import { useState, useRouter } from '@/lib/router';
import { addCapsuleClient } from '@/lib/capsules';
import {
  PixelCard,
  PixelInput,
  PixelTextarea,
  PixelButton,
} from '@/components/pixel';
import { Container } from '@/components/layout/Container';

// 简化版 useRouter（实际项目里用 next/navigation）
import { useRouter as nextUseRouter } from 'next/navigation';

export default function NewCapsulePage() {
  const router = nextUseRouter();
  const [title, setTitle] = useState('');
  const [content, setContent] = useState('');
  const [unlockAt, setUnlockAt] = useState('');
  const [error, setError] = useState('');

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    setError('');

    // 校验
    if (!title.trim()) return setError('标题不能为空');
    if (!content.trim()) return setError('内容不能为空');
    if (!unlockAt) return setError('请选择解锁时间');

    const unlockDate = new Date(unlockAt);
    if (unlockDate <= new Date()) {
      return setError('解锁时间必须在未来');
    }

    addCapsuleClient({
      title: title.trim(),
      content: content.trim(),
      unlockAt: unlockDate.toISOString(),
    });

    router.push('/capsules');
  };

  return (
    <Container>
      <h1 className="text-4xl font-bold mb-6">[ 写时光胶囊 ]</h1>

      <PixelCard>
        <form onSubmit={handleSubmit} className="p-4 flex flex-col gap-4">
          <PixelInput
            label="胶囊标题"
            placeholder="比如：致 2027 年的我"
            value={title}
            onChange={(e) => setTitle(e.target.value)}
            maxLength={100}
            required
          />

          <PixelTextarea
            label="胶囊内容"
            placeholder="写下你想说的话..."
            value={content}
            onChange={(e) => setContent(e.target.value)}
            rows={8}
            maxLength={10000}
            required
          />

          <PixelInput
            label="解锁时间"
            type="datetime-local"
            value={unlockAt}
            onChange={(e) => setUnlockAt(e.target.value)}
            min={new Date(Date.now() + 60_000).toISOString().slice(0, 16)}
            required
            helperText="必须是未来的时间"
          />

          {error && (
            <p className="text-sm">[!] {error}</p>
          )}

          <div className="flex gap-2 flex-wrap">
            <PixelButton type="submit" variant="primary">
              [封存]
            </PixelButton>
            <PixelButton type="button" onClick={() => router.back()}>
              取消
            </PixelButton>
          </div>
        </form>
      </PixelCard>

      <p className="text-sm mt-4 italic">
        💡 提示：胶囊内容存在你的浏览器本地，换设备或清除缓存会丢失。
        重要内容记得手动备份。
      </p>
    </Container>
  );
}
```

## 六、胶囊详情页（可选）

新建 `src/app/capsules/[id]/page.tsx`：

```tsx
'use client';

import { useEffect, useState } from 'react';
import { notFound, useRouter } from 'next/navigation';
import Link from 'next/link';
import { getAllCapsulesClient, deleteCapsuleClient, withStatus } from '@/lib/capsules';
import { PixelCard, PixelButton, PixelModal } from '@/components/pixel';
import { Container } from '@/components/layout/Container';
import { formatDate } from '@/lib/date';

export default function CapsuleDetailPage({ params }: { params: Promise<{ id: string }> }) {
  const router = useRouter();
  const [id, setId] = useState('');
  const [capsule, setCapsule] = useState<ReturnType<typeof withStatus> | null>(null);
  const [showDelete, setShowDelete] = useState(false);

  useEffect(() => {
    params.then((p) => setId(p.id));
  }, [params]);

  useEffect(() => {
    if (!id) return;
    const all = getAllCapsulesClient();
    const found = all.find((c) => c.id === id);
    if (!found) {
      notFound();
      return;
    }
    setCapsule(withStatus(found));
  }, [id]);

  if (!capsule) return <Container><p>加载中...</p></Container>;

  const handleDelete = () => {
    deleteCapsuleClient(capsule.id);
    router.push('/capsules');
  };

  return (
    <Container>
      <PixelCard>
        <div className="p-6 flex flex-col gap-4">
          <div className="flex items-start justify-between gap-4">
            <h1 className="text-4xl font-bold">{capsule.title}</h1>
            <span className="text-3xl">
              {capsule.unlocked ? '[💌]' : '[🔒]'}
            </span>
          </div>

          <p className="text-sm">
            解锁时间：{formatDate(capsule.unlockAt)}
          </p>

          {capsule.unlocked ? (
            <div className="border-t-2 border-fg pt-4 mt-2">
              <p className="whitespace-pre-wrap leading-relaxed">{capsule.content}</p>
            </div>
          ) : (
            <div className="border-t-2 border-fg pt-4 mt-2 text-center">
              <p className="text-2xl mb-2">🔒</p>
              <p>这封信还没到时间</p>
              <p className="text-sm mt-2">
                还有 <strong>{capsule.daysUntilUnlock}</strong> 天
              </p>
              <p className="text-xs mt-4 opacity-70 italic">
                （技术上讲，你可以改电脑时间来看，但那就违背了写信的初衷）
              </p>
            </div>
          )}

          <p className="text-xs opacity-70">
            创建于 {formatDate(capsule.createdAt)}
          </p>
        </div>
      </PixelCard>

      <div className="mt-4 flex gap-2 flex-wrap">
        <Link href="/capsules">
          <PixelButton>← 返回列表</PixelButton>
        </Link>
        <PixelButton onClick={() => setShowDelete(true)}>
          删除
        </PixelButton>
      </div>

      <PixelModal
        open={showDelete}
        onClose={() => setShowDelete(false)}
        title="确认删除？"
      >
        <p className="mb-4">删除后无法恢复。</p>
        <div className="flex gap-2 justify-end">
          <PixelButton onClick={() => setShowDelete(false)}>取消</PixelButton>
          <PixelButton variant="primary" onClick={handleDelete}>
            确认删除
          </PixelButton>
        </div>
      </PixelModal>
    </Container>
  );
}
```

## 八、跨设备方案（可选）

> 如果你想在手机上也能看到胶囊，**用 JSON + GitHub API**。

### 1. 建 `data/capsules.json`

```json
[]
```

> 先建空数组。

### 2. 准备 GitHub Token

1. GitHub → Settings → Developer settings → Personal access tokens → Tokens (classic)
2. Generate new token → 勾 `repo` → 生成
3. 复制 token

### 3. 加环境变量 `.env.local`

```env
GITHUB_TOKEN=ghp_xxxxxxxxxxxx
GITHUB_REPO=your-name/your-repo
```

### 4. 装 octokit

```bash
pnpm add @octokit/rest
```

### 5. API 路由 `src/app/api/capsules/route.ts`

```ts
import { NextResponse } from 'next/server';
import { Octokit } from '@octokit/rest';

const octokit = new Octokit({ auth: process.env.GITHUB_TOKEN });
const [owner, repo] = (process.env.GITHUB_REPO ?? '').split('/');
const FILE_PATH = 'data/capsules.json';

async function readCapsules(): Promise<unknown[]> {
  try {
    const { data } = await octokit.repos.getContent({
      owner,
      repo,
      path: FILE_PATH,
    });
    if (Array.isArray(data)) return [];
    const content = Buffer.from(data.content, 'base64').toString('utf-8');
    return JSON.parse(content);
  } catch {
    return [];
  }
}

async function writeCapsules(capsules: unknown[]): Promise<void> {
  const { data: current } = await octokit.repos.getContent({
    owner,
    repo,
    path: FILE_PATH,
  });
  const content = Buffer.from(JSON.stringify(capsules, null, 2)).toString('base64');
  await octokit.repos.createOrUpdateFileContents({
    owner,
    repo,
    path: FILE_PATH,
    message: 'chore: 更新 capsules.json',
    content,
    sha: Array.isArray(current) ? undefined : current.sha,
  });
}

export async function GET() {
  const capsules = await readCapsules();
  return NextResponse.json(capsules);
}

export async function POST(request: Request) {
  const body = await request.json();
  const capsules = await readCapsules();
  capsules.push(body);
  await writeCapsules(capsules);
  return NextResponse.json({ success: true });
}

export async function DELETE(request: Request) {
  const { id } = await request.json();
  const capsules = await readCapsules();
  const filtered = capsules.filter((c: { id: string }) => c.id !== id);
  await writeCapsules(filtered);
  return NextResponse.json({ success: true });
}
```

### 6. 改 `src/lib/capsules.ts` 走 API

```ts
// 把 client 版本的函数改成走 API
export async function getAllCapsulesApi(): Promise<Capsule[]> {
  const res = await fetch('/api/capsules');
  return res.json();
}

export async function addCapsuleApi(input: Omit<Capsule, 'id' | 'createdAt'>) {
  const newCapsule = {
    id: cryptoRandomUUID(),
    createdAt: new Date().toISOString(),
    ...input,
  };
  await fetch('/api/capsules', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify(newCapsule),
  });
  return newCapsule;
}

export async function deleteCapsuleApi(id: string) {
  await fetch('/api/capsules', {
    method: 'DELETE',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ id }),
  });
}
```
