## 二、文件结构

```
src/
├── app/about/
│   └── page.tsx        ← 页面
└── data/
    └── about.json      ← 内容（也可以用 MDX）
```

**两种写法**：
- **A. JSON**：简单，结构化字段
- **B. MDX**：自己写，能用 Markdown 格式

我们用 **A（JSON）**，简单够用。

新建 `src/types/about.ts`：

```ts
export type Contact = {
  type: 'email' | 'github' | 'twitter' | 'bilibili' | 'weibo' | 'netease' | 'instagram' | 'website' | 'rss' | string;
  label: string;
  url: string;
};

export type Tool = {
  category: string;
  name: string;
};

export type Book = {
  title: string;
  status: '在读' | '读过' | '想读';
};

export type AboutData = {
  name: string;
  avatar: string;
  tagline: string;
  city?: string;
  job?: string;
  bio: string[];
  contacts: Contact[];
  tools?: Tool[];
  books?: Book[];
  music?: string[];
};
```

## 四、读取数据

新建 `src/lib/about.ts`：

```ts
import aboutData from '@/../data/about.json';
import type { AboutData } from '@/types/about';

const data = aboutData as AboutData;

export function getAbout(): AboutData {
  return data;
}
```

> `import aboutData from '@/../data/about.json'` —— `@/` 是 `src/`，所以 `@/../` 是项目根目录，`@/../data/` 就是 `/data/`。

## 五、关于我页面

新建 `src/app/about/page.tsx`：

```tsx
import Image from 'next/image';
import Link from 'next/link';
import { getAbout } from '@/lib/about';
import { PixelCard } from '@/components/pixel';
import { Container } from '@/components/layout/Container';

export const metadata = {
  title: '关于我 · PixelBlog',
  description: '关于这个博客和它的主人',
};

export default function AboutPage() {
  const about = getAbout();

  return (
    <Container>
      {/* 头部：头像 + 名字 */}
      <header className="flex items-start gap-6 mb-8 flex-wrap">
        <Image
          src={about.avatar}
          alt={about.name}
          width={128}
          height={128}
          className="border-2 border-fg shadow-pixel"
          unoptimized  // 像素图不要 Next.js 的图片优化（会模糊）
        />
        <div className="flex-1 min-w-0">
          <h1 className="text-4xl font-bold">{about.name}</h1>
          <p className="mt-2">{about.tagline}</p>
          {(about.city || about.job) && (
            <p className="text-sm mt-2">
              {about.city && <>[{about.city}] </>}
              {about.job && <>· [{about.job}] </>}
            </p>
          )}
        </div>
      </header>

      {/* 自我介绍 */}
      <PixelCard className="mb-6">
        <h2 className="text-2xl font-bold p-4 border-b-2 border-fg">关于我</h2>
        <div className="p-4 leading-relaxed">
          {about.bio.map((line, i) => (
            <p key={i} className="my-2">
              {line}
            </p>
          ))}
        </div>
      </PixelCard>

      {/* 联系方式 */}
      {about.contacts.length > 0 && (
        <PixelCard className="mb-6">
          <h2 className="text-2xl font-bold p-4 border-b-2 border-fg">联系方式</h2>
          <ul className="p-4 flex flex-wrap gap-3">
            {about.contacts.map((c) => (
              <li key={c.url}>
                <ContactLink contact={c} />
              </li>
            ))}
          </ul>
        </PixelCard>
      )}

      {/* 我的工具 */}
      {about.tools && about.tools.length > 0 && (
        <PixelCard className="mb-6">
          <h2 className="text-2xl font-bold p-4 border-b-2 border-fg">我的工具</h2>
          <ul className="p-4 grid grid-cols-1 sm:grid-cols-2 gap-2">
            {about.tools.map((t, i) => (
              <li key={i} className="border border-fg p-2">
                [{t.category}] {t.name}
              </li>
            ))}
          </ul>
        </PixelCard>
      )}

      {/* 在读 / 读过 */}
      {about.books && about.books.length > 0 && (
        <PixelCard className="mb-6">
          <h2 className="text-2xl font-bold p-4 border-b-2 border-fg">书</h2>
          <ul className="p-4 flex flex-col gap-1">
            {about.books.map((b, i) => (
              <li key={i}>
                [{b.status}] {b.title}
              </li>
            ))}
          </ul>
        </PixelCard>
      )}

      {/* 音乐 */}
      {about.music && about.music.length > 0 && (
        <PixelCard className="mb-6">
          <h2 className="text-2xl font-bold p-4 border-b-2 border-fg">最近听</h2>
          <ul className="p-4 flex flex-col gap-1">
            {about.music.map((m, i) => (
              <li key={i}>· {m}</li>
            ))}
          </ul>
        </PixelCard>
      )}
    </Container>
  );
}

function ContactLink({ contact }: { contact: { label: string; url: string } }) {
  const isExternal = contact.url.startsWith('http');
  return (
    <Link
      href={contact.url}
      target={isExternal ? '_blank' : undefined}
      rel={isExternal ? 'noopener noreferrer' : undefined}
      className="
        no-underline
        inline-flex items-center
        border-2 border-fg px-3 py-1
        hover:bg-fg hover:text-bg
        transition-none
      "
    >
      [{contact.label}]
    </Link>
  );
}
```

## 六、像素头像怎么做

### 方案 1：自己画

工具：**Aseprite**（付费，30 美元）或 **Piskel**（免费，网页版）

- 画一个 32x32 或 64x64 的像素角色
- 导出 PNG
- 放到 `public/sprites/me.png`

### 方案 2：AI 生成

提示词示例（给 Midjourney / DALL-E / SD）：

```
A pixel art portrait of a young person, 1-bit style, only black and white, 64x64 pixels, no anti-aliasing, retro game character style, transparent background
```

保存到 `public/sprites/me.png`。

### 方案 3：临时占位

先用 Next.js 默认图片占位：

```tsx
<div className="w-32 h-32 border-2 border-fg shadow-pixel bg-bg flex items-center justify-center text-6xl">
  [·]
</div>
```
