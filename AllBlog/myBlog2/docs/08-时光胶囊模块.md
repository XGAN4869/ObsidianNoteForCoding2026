# 08 · 时光胶囊模块（手把手）

> 胶囊 = 随笔的 CRUD 套路 + 一个"到期才能打开"的时间魔法。
> 核心就一件事：**`openAt` 在未来时封住，到了时间才放行。**

---

## 一、API 层

### 1. 列表 + 新建：`app/api/capsules/route.ts`

```ts
// app/api/capsules/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";
import { requireAuth } from "@/lib/auth";

// GET 胶囊列表（前端会自己区分密封/可打开）
export async function GET() {
  const now = new Date();
  const capsules = await db.capsule.findMany({
    orderBy: { createdAt: "desc" },
  });

  // 后端计算出"能否打开"，不要前端用本地时间判断
  const data = capsules.map((c) => ({
    ...c,
    canOpen: !c.isOpened && now >= c.openAt,
  }));

  return NextResponse.json({ data });
}

// POST 埋胶囊
export async function POST(req: Request) {
  if (!(await requireAuth(req))) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }

  const { title, content, mood = "CALM", openAt } = await req.json();

  // 校验：内容不为空，且打开时间必须晚于现在
  if (!title?.trim() || !content?.trim()) {
    return NextResponse.json({ error: "内容不能为空" }, { status: 400 });
  }
  const openDate = new Date(openAt);
  if (isNaN(openDate.getTime()) || openDate <= new Date()) {
    return NextResponse.json(
      { error: "打开时间必须是未来" },
      { status: 400 }
    );
  }

  const capsule = await db.capsule.create({
    data: { title, content, mood, openAt: openDate },
  });

  return NextResponse.json({ data: capsule }, { status: 201 });
}
```

### 2. 打开 + 删除：`app/api/capsules/[id]/route.ts`

```ts
// app/api/capsules/[id]/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";
import { requireAuth } from "@/lib/auth";

// DELETE 删除胶囊
export async function DELETE(
  req: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  if (!(await requireAuth(req))) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }
  const { id } = await params;
  await db.capsule.delete({ where: { id } });
  return NextResponse.json({ data: { ok: true } });
}
```

### 3. 打开动作：`app/api/capsules/[id]/open/route.ts`

```ts
// app/api/capsules/[id]/open/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";
import { requireAuth } from "@/lib/auth";

export async function POST(
  req: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  if (!(await requireAuth(req))) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }

  const { id } = await params;
  const capsule = await db.capsule.findUnique({ where: { id } });

  if (!capsule) {
    return NextResponse.json({ error: "胶囊不存在" }, { status: 404 });
  }

  // 关键校验：没到期不让开（防止绕过前端直接调 API）
  if (new Date() < capsule.openAt) {
    return NextResponse.json({ error: "还没到打开的时间" }, { status: 403 });
  }

  const updated = await db.capsule.update({
    where: { id },
    data: { isOpened: true, openedAt: new Date() },
  });

  return NextResponse.json({ data: updated });
}
```

> 记住：**"未到期不能开"这个规则必须在后端再校验一次**，前端只是体验层。

## 二、前端：胶囊列表页

创建 `app/capsule/page.tsx`：

```tsx
// app/capsule/page.tsx
"use client";

import { useEffect, useState } from "react";
import MoodIcon from "@/components/mood/MoodIcon";
import type { Mood } from "@/lib/moods";

type Capsule = {
  id: string;
  title: string;
  content: string;
  mood: Mood;
  openAt: string;
  isOpened: boolean;
  canOpen: boolean;
  openedAt: string | null;
  createdAt: string;
};

function daysUntil(date: string) {
  const diff = new Date(date).getTime() - Date.now();
  return Math.ceil(diff / (1000 * 60 * 60 * 24));
}

export default function CapsulePage() {
  const [capsules, setCapsules] = useState<Capsule[]>([]);
  const [showForm, setShowForm] = useState(false);

  async function load() {
    const res = await fetch("/api/capsules");
    const d = await res.json();
    setCapsules(d.data);
  }

  useEffect(() => {
    load();
  }, []);

  async function openCapsule(id: string) {
    const res = await fetch(`/api/capsules/${id}/open`, { method: "POST" });
    if (res.ok) {
      load(); // 重新加载列表
    } else {
      const d = await res.json();
      alert(d.error);
    }
  }

  return (
    <div className="max-w-2xl mx-auto p-6 space-y-4">
      <div className="flex items-center justify-between">
        <h1 className="font-hand text-3xl">时光胶囊</h1>
        <button
          onClick={() => setShowForm(!showForm)}
          className="bg-[var(--ink)] text-[var(--bg)] border-2 border-[var(--ink)] rounded-[12px_4px_14px_6px/6px_14px_4px_12px] px-4 py-2 shadow-[3px_3px_0_var(--ink)]"
        >
          + 埋一个
        </button>
      </div>

      {showForm && <CapsuleForm onSaved={() => { setShowForm(false); load(); }} />}

      {capsules.length === 0 && (
        <p className="text-[var(--ink-soft)]">还没有胶囊，给未来的自己写一封信吧。</p>
      )}

      {capsules.map((c) => (
        <article key={c.id} className="hand-drawn p-5">
          <div className="flex items-center gap-3">
            <MoodIcon mood={c.mood} size={28} />
            <h2 className="font-bold text-lg flex-1">{c.title}</h2>
            {c.isOpened ? (
              <span className="text-sm text-[var(--ink-soft)]">🔓 已打开</span>
            ) : c.canOpen ? (
              <button
                onClick={() => openCapsule(c.id)}
                className="text-sm border-2 border-[var(--accent)] text-[var(--ink)] rounded-lg px-3 py-1 hover:bg-[var(--accent)]"
              >
                🔓 打开
              </button>
            ) : (
              <span className="text-sm text-[var(--ink-soft)]">
                🔒 还有 {daysUntil(c.openAt)} 天
              </span>
            )}
          </div>

          {/* 已打开的显示内容，未打开显示占位 */}
          {c.isOpened ? (
            <div className="mt-3 whitespace-pre-wrap leading-relaxed">{c.content}</div>
          ) : (
            <p className="mt-3 text-[var(--ink-soft)] text-sm">
              （内容被封存，{new Date(c.openAt).toLocaleDateString("zh-CN")} 才能打开）
            </p>
          )}
        </article>
      ))}
    </div>
  );
}

function CapsuleForm({ onSaved }: { onSaved: () => void }) {
  const [title, setTitle] = useState("");
  const [content, setContent] = useState("");
  const [openAt, setOpenAt] = useState("");
  const [saving, setSaving] = useState(false);

  async function submit(e: React.FormEvent) {
    e.preventDefault();
    setSaving(true);
    const res = await fetch("/api/capsules", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ title, content, openAt: new Date(openAt).toISOString() }),
    });
    setSaving(false);
    if (res.ok) {
      onSaved();
    } else {
      const d = await res.json();
      alert(d.error);
    }
  }

  return (
    <form onSubmit={submit} className="hand-drawn p-5 space-y-3">
      <input
        value={title}
        onChange={(e) => setTitle(e.target.value)}
        placeholder="标题，如：写给 2027 年的我"
        className="w-full bg-[var(--bg-muted)] border-2 border-[var(--ink)] rounded-lg px-3 py-2"
      />
      <textarea
        value={content}
        onChange={(e) => setContent(e.target.value)}
        placeholder="想对未来的自己说什么……"
        rows={5}
        className="w-full bg-[var(--bg-muted)] border-2 border-[var(--ink)] rounded-lg px-3 py-2"
      />
      <div>
        <label className="text-sm text-[var(--ink-soft)]">什么时候打开：</label>
        <input
          type="datetime-local"
          value={openAt}
          onChange={(e) => setOpenAt(e.target.value)}
          min={new Date().toISOString().slice(0, 16)} // 不能选过去
          className="ml-2 bg-[var(--bg-muted)] border-2 border-[var(--ink)] rounded-lg px-3 py-2"
        />
      </div>
      <button
        type="submit"
        disabled={saving}
        className="bg-[var(--ink)] text-[var(--bg)] border-2 border-[var(--ink)] rounded-[12px_4px_14px_6px/6px_14px_4px_12px] px-5 py-2"
      >
        {saving ? "埋入中…" : "埋入胶囊"}
      </button>
    </form>
  );
}
```

## 三、时区坑（师傅重点提醒）

1. `datetime-local` 返回的是本地时间字符串（如 `2027-01-01T00:00`），没有时区信息。
2. 转成 `Date` 时用 `new Date(openAt)` 会按本地时区解析，正确。
3. 存进数据库后是 UTC，取出来展示时用 `toLocaleDateString("zh-CN")` 会自动转回本地。
4. **"还有 N 天"的计算**要用时间戳差值，不要手动算年月日（会踩闰年、月底的坑）。
