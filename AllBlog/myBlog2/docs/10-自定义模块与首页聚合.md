# 10 · 自定义模块与首页聚合（手把手）

> 这一章做两件事：1) 让首页变成"白空间"（聚合一切）；2) 支持添加自定义小模块（时钟、番茄钟、一句话等）。
> 这是让博客"活过来"的一章。

## 一、自定义模块的 API

### `app/api/widgets/route.ts`（列表 + 新建）

```ts
// app/api/widgets/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";
import { requireAuth } from "@/lib/auth";

export async function GET() {
  const widgets = await db.widget.findMany({
    where: { enabled: true },
    orderBy: { position: "asc" },
  });
  return NextResponse.json({ data: widgets });
}

export async function POST(req: Request) {
  if (!(await requireAuth(req))) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }
  const { type, config = {} } = await req.json();
  const widget = await db.widget.create({
    data: { type, config: JSON.stringify(config) },
  });
  return NextResponse.json({ data: widget }, { status: 201 });
}
```

### `app/api/widgets/[id]/route.ts`（更新 + 删除）

```ts
// app/api/widgets/[id]/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";
import { requireAuth } from "@/lib/auth";

export async function PATCH(
  req: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  if (!(await requireAuth(req))) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }
  const { id } = await params;
  const body = await req.json();
  const widget = await db.widget.update({
    where: { id },
    data: {
      ...(body.config !== undefined && { config: JSON.stringify(body.config) }),
      ...(body.position !== undefined && { position: body.position }),
      ...(body.enabled !== undefined && { enabled: body.enabled }),
    },
  });
  return NextResponse.json({ data: widget });
}

export async function DELETE(
  req: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  if (!(await requireAuth(req))) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }
  const { id } = await params;
  await db.widget.delete({ where: { id } });
  return NextResponse.json({ data: { ok: true } });
}
```

## 二、模块组件（以时钟 + 番茄钟为例）

每个模块是一个独立组件，接收 `widget` 数据。先做两个最实用的：

### 1. 时钟模块

```tsx
// components/widgets/ClockWidget.tsx
"use client";

import { useEffect, useState } from "react";

export default function ClockWidget() {
  const [now, setNow] = useState(new Date());

  useEffect(() => {
    // 每秒更新，记得清理定时器
    const timer = setInterval(() => setNow(new Date()), 1000);
    return () => clearInterval(timer); // 关键：防内存泄漏
  }, []);

  const hh = String(now.getHours()).padStart(2, "0");
  const mm = String(now.getMinutes()).padStart(2, "0");
  const ss = String(now.getSeconds()).padStart(2, "0");

  return (
    <div className="text-center py-4">
      <div className="font-caveat text-5xl tracking-widest">
        {hh}:{mm}:{ss}
      </div>
      <div className="text-xs text-[var(--ink-soft)] mt-1">
        {now.toLocaleDateString("zh-CN", { weekday: "long", month: "long", day: "numeric" })}
      </div>
    </div>
  );
}
```

### 2. 番茄钟模块

```tsx
// components/widgets/PomodoroWidget.tsx
"use client";

import { useEffect, useState } from "react";

export default function PomodoroWidget({ config }: { config: { duration?: number } }) {
  const duration = config.duration ?? 25; // 默认 25 分钟
  const [seconds, setSeconds] = useState(duration * 60);
  const [running, setRunning] = useState(false);

  useEffect(() => {
    if (!running) return;
    const timer = setInterval(() => {
      setSeconds((s) => {
        if (s <= 1) {
          setRunning(false);
          return 0;
        }
        return s - 1;
      });
    }, 1000);
    return () => clearInterval(timer);
  }, [running]);

  const mm = String(Math.floor(seconds / 60)).padStart(2, "0");
  const ss = String(seconds % 60).padStart(2, "0");

  return (
    <div className="text-center py-4 space-y-2">
      <div className="font-caveat text-4xl">{mm}:{ss}</div>
      <button
        onClick={() => setRunning((r) => !r)}
        className="text-sm border-2 border-[var(--ink)] rounded-lg px-3 py-1"
      >
        {running ? "暂停" : "开始"}
      </button>
      <button
        onClick={() => setSeconds(duration * 60)}
        className="text-sm border-2 border-[var(--ink)] rounded-lg px-3 py-1 ml-2"
      >
        重置
      </button>
    </div>
  );
}
```

## 三、模块注册表（方便扩展）

```ts
// lib/widgets.tsx
import ClockWidget from "@/components/widgets/ClockWidget";
import PomodoroWidget from "@/components/widgets/PomodoroWidget";

// 类型 → 组件 + 元信息
export const WIDGET_TYPES = {
  clock: {
    label: "时钟",
    icon: "🕐",
    component: ClockWidget,
    defaultConfig: {},
  },
  pomodoro: {
    label: "番茄钟",
    icon: "🍅",
    component: PomodoroWidget,
    defaultConfig: { duration: 25 },
  },
  // 以后加新模块，只需在这里注册
  // quote: { label: "一句话", icon: "💬", component: QuoteWidget, defaultConfig: {} },
} as const;

export type WidgetType = keyof typeof WIDGET_TYPES;
```

> 这个"注册表"模式是重点：以后加新模块，写一个组件 + 在这里登记一行即可，不用改别处。

## 四、首页（白空间）聚合

创建/改造 `app/page.tsx`：

```tsx
// app/page.tsx
"use client";

import { useEffect, useState } from "react";
import Link from "next/link";
import FloatingSquares from "@/components/layout/FloatingSquares";
import MoodIcon from "@/components/mood/MoodIcon";
import { MOOD_LIST } from "@/lib/moods";
import { WIDGET_TYPES } from "@/lib/widgets";

type Widget = { id: string; type: string; config: string };

export default function HomePage() {
  const [widgets, setWidgets] = useState<Widget[]>([]);
  const [todoCount, setTodoCount] = useState(0);
  const [memory, setMemory] = useState<string | null>(null);
  const [mood, setMood] = useState("CALM");

  useEffect(() => {
    fetch("/api/widgets").then((r) => r.json()).then((d) => setWidgets(d.data));
    fetch("/api/todos").then((r) => r.json()).then((d) =>
      setTodoCount(d.data.filter((t: any) => !t.done).length)
    );
    fetch("/api/random-note").then((r) => r.json()).then((d) => setMemory(d.data));
  }, []);

  return (
    <div className="min-h-screen relative">
      <FloatingSquares />

      <main className="max-w-3xl mx-auto p-6 pt-16 space-y-8 relative">
        {/* 今天的心情 */}
        <section className="text-center space-y-3">
          <p className="text-[var(--ink-soft)]">今天的心情</p>
          <div className="flex justify-center gap-2">
            {MOOD_LIST.map((m) => (
              <button
                key={m.key}
                onClick={() => setMood(m.key)}
                className={`p-2 rounded-full border-2 transition-all ${
                  mood === m.key
                    ? "border-[var(--ink)] scale-110"
                    : "border-transparent opacity-60"
                }`}
                title={m.label}
              >
                <MoodIcon mood={m.key} size={32} />
              </button>
            ))}
          </div>
        </section>

        {/* 三扇门 */}
        <section className="grid grid-cols-3 gap-4">
          <Link href="/notes" className="hand-drawn hand-drawn-hover p-6 text-center">
            <div className="text-3xl">📝</div>
            <div className="font-hand text-xl mt-2">随笔</div>
          </Link>
          <Link href="/capsule" className="hand-drawn hand-drawn-hover p-6 text-center">
            <div className="text-3xl">⏳</div>
            <div className="font-hand text-xl mt-2">胶囊</div>
          </Link>
          <Link href="/todos" className="hand-drawn hand-drawn-hover p-6 text-center">
            <div className="text-3xl">✅</div>
            <div className="font-hand text-xl mt-2">待办</div>
          </Link>
        </section>

        {/* 今日待办精简 */}
        <section className="hand-drawn p-5">
          <div className="flex items-center justify-between">
            <h2 className="font-hand text-xl">今日待办</h2>
            <Link href="/todos" className="text-sm text-[var(--ink-soft)]">
              还有 {todoCount} 件 →
            </Link>
          </div>
        </section>

        {/* 随机回忆 */}
        {memory && (
          <section className="hand-drawn p-5 text-center">
            <p className="text-xs text-[var(--ink-soft)] mb-2">突然想起……</p>
            <p className="italic leading-relaxed">{memory}</p>
          </section>
        )}

        {/* 自定义模块 */}
        {widgets.length > 0 && (
          <section className="grid grid-cols-2 gap-4">
            {widgets.map((w) => {
              const type = WIDGET_TYPES[w.type as keyof typeof WIDGET_TYPES];
              if (!type) return null;
              const Comp = type.component;
              const config = w.config ? JSON.parse(w.config) : {};
              return (
                <div key={w.id} className="hand-drawn p-4">
                  <Comp config={config} />
                </div>
              );
            })}
          </section>
        )}
      </main>
    </div>
  );
}
```

## 五、导航栏（统一入口）

创建 `components/layout/Navbar.tsx`：

```tsx
// components/layout/Navbar.tsx
"use client";

import Link from "next/link";
import { usePathname } from "next/navigation";

const links = [
  { href: "/", label: "白空间" },
  { href: "/notes", label: "随笔" },
  { href: "/capsule", label: "胶囊" },
  { href: "/todos", label: "待办" },
  { href: "/board", label: "布置" },
];

export default function Navbar() {
  const pathname = usePathname();

  return (
    <header className="sticky top-0 z-10 border-b-2 border-[var(--ink)] bg-[var(--bg)]/90 backdrop-blur">
      <div className="max-w-3xl mx-auto px-6 py-3 flex items-center justify-between">
        <Link href="/" className="font-hand text-2xl">
          ✦ 我的空间
        </Link>
        <nav className="flex gap-4">
          {links.map((l) => (
            <Link
              key={l.href}
              href={l.href}
              className={`text-sm ${
                pathname === l.href
                  ? "font-bold underline underline-offset-4"
                  : "text-[var(--ink-soft)] hover:text-[var(--ink)]"
              }`}
            >
              {l.label}
            </Link>
          ))}
        </nav>
      </div>
    </header>
  );
}
```

## 六、布置页（添加/删除模块）

创建 `app/board/page.tsx`：

```tsx
// app/board/page.tsx
"use client";

import { useEffect, useState } from "react";
import { WIDGET_TYPES, type WidgetType } from "@/lib/widgets";

type Widget = { id: string; type: string; config: string };

export default function BoardPage() {
  const [widgets, setWidgets] = useState<Widget[]>([]);

  async function load() {
    const res = await fetch("/api/widgets");
    const d = await res.json();
    setWidgets(d.data);
  }

  useEffect(() => {
    load();
  }, []);

  async function addWidget(type: WidgetType) {
    const def = WIDGET_TYPES[type];
    await fetch("/api/widgets", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ type, config: def.defaultConfig }),
    });
    load();
  }

  async function removeWidget(id: string) {
    await fetch(`/api/widgets/${id}`, { method: "DELETE" });
    load();
  }

  return (
    <div className="max-w-2xl mx-auto p-6 space-y-4">
      <h1 className="font-hand text-3xl">布置我的空间</h1>

      {/* 可添加的模块类型 */}
      <div className="hand-drawn p-4">
        <p className="text-sm text-[var(--ink-soft)] mb-2">添加模块：</p>
        <div className="flex gap-2 flex-wrap">
          {(Object.keys(WIDGET_TYPES) as WidgetType[]).map((type) => (
            <button
              key={type}
              onClick={() => addWidget(type)}
              className="border-2 border-[var(--ink)] rounded-lg px-3 py-1 text-sm hover:bg-[var(--bg-muted)]"
            >
              {WIDGET_TYPES[type].icon} {WIDGET_TYPES[type].label}
            </button>
          ))}
        </div>
      </div>

      {/* 已添加的模块 */}
      {widgets.length === 0 ? (
        <p className="text-[var(--ink-soft)]">还没有模块，加一个时钟或番茄钟吧。</p>
      ) : (
        <div className="grid grid-cols-2 gap-4">
          {widgets.map((w) => (
            <div key={w.id} className="hand-drawn p-4 relative">
              <button
                onClick={() => removeWidget(w.id)}
                className="absolute top-2 right-2 text-[var(--danger)] text-xs"
              >
                ✕
              </button>
              <div className="opacity-60 pointer-events-none">
                {/* 只预览，不真正计时，简化处理 */}
                <p className="text-center font-hand text-xl">
                  {WIDGET_TYPES[w.type as WidgetType]?.label}
                </p>
              </div>
            </div>
          ))}
        </div>
      )}
    </div>
  );
}
```
