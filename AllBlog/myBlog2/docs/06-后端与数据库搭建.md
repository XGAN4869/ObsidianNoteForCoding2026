## 一、写 Prisma Schema

打开 `prisma/schema.prisma`，**完整替换**为（就是 02 章设计好的模型）：

```prisma
generator client {
  provider = "prisma-client-js"
}

datasource db {
  provider = "sqlite"
  url      = env("DATABASE_URL")
}

model User {
  id        String   @id @default(cuid())
  name      String
  password  String
  createdAt DateTime @default(now())
}

enum Mood {
  CALM
  HAPPY
  SAD
  ANGRY
  AFRAID
  TIRED
  LONELY
}

model Note {
  id        String   @id @default(cuid())
  title     String
  content   String
  mood      Mood     @default(CALM)
  tags      Tag[]
  isDraft   Boolean  @default(false)
  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt
}

model Tag {
  id     String  @id @default(cuid())
  name   String  @unique
  notes  Note[]
}

model Capsule {
  id        String    @id @default(cuid())
  title     String
  content   String
  mood      Mood      @default(CALM)
  openAt    DateTime
  isOpened  Boolean   @default(false)
  createdAt DateTime  @default(now())
  openedAt  DateTime?
}

model Todo {
  id          String    @id @default(cuid())
  title       String
  done        Boolean   @default(false)
  dueDate     DateTime?
  order       Int       @default(0)
  createdAt   DateTime  @default(now())
  completedAt DateTime?
}

model MoodLog {
  id    String   @id @default(cuid())
  mood  Mood
  date  DateTime @unique
  note  String?
}

model Widget {
  id       String  @id @default(cuid())
  type     String
  title    String?
  config   String
  position Int     @default(0)
  enabled  Boolean @default(true)
}

model Setting {
  key   String @id
  value String
}
```

保存后运行迁移，生成数据库：

```bash
npx prisma migrate dev --name init
```

> 这条命令会：1) 根据 schema 生成 SQLite 数据库文件；2) 生成 Prisma Client（可调用的代码）。
> 以后每次改 schema，都要再跑一次 `npx prisma migrate dev --name 描述`。

## 二、Prisma 客户端单例

创建 `lib/db.ts`：

```ts
// lib/db.ts
import { PrismaClient } from "@prisma/client";

// 开发时热重载会重复创建连接，用 globalThis 缓存单例
const globalForPrisma = globalThis as unknown as { prisma?: PrismaClient };

export const db = globalForPrisma.prisma ?? new PrismaClient();

if (process.env.NODE_ENV !== "production") {
  globalForPrisma.prisma = db;
}
```

> 之后所有 API 里 `import { db } from "@/lib/db"` 就能操作数据库。

## 三、登录认证

### 1. 设置密码

打开 `.env`，加两行：

```env
BLOG_PASSWORD=你的密码，随便设一个
JWT_SECRET=一串很长的随机字符串，比如随便打 30 个字母数字
```

> `JWT_SECRET` 用来签名登录凭证。可以随便编一串，但要足够长且保密。

### 2. 登录 API

创建 `app/api/auth/login/route.ts`：

```ts
// app/api/auth/login/route.ts
import { NextResponse } from "next/server";
import { SignJWT } from "jose";
import { compare } from "bcryptjs";
import { db } from "@/lib/db";

const secret = new TextEncoder().encode(process.env.JWT_SECRET!);

export async function POST(req: Request) {
  const { password } = await req.json();

  // 查数据库里的用户（第一个用户就是你自己）
  const user = await db.user.findFirst();
  if (!user) {
    return NextResponse.json(
      { error: "还没有用户，请先初始化" },
      { status: 500 }
    );
  }

  // 比对密码（bcrypt 哈希比对）
  const ok = await compare(password, user.password);
  if (!ok) {
    return NextResponse.json({ error: "密码不对" }, { status: 401 });
  }

  // 签发 30 天有效的 JWT
  const token = await new SignJWT({ uid: user.id })
    .setProtectedHeader({ alg: "HS256" })
    .setExpirationTime("30d")
    .sign(secret);

  const res = NextResponse.json({ data: { ok: true } });
  // 写入 httpOnly Cookie（JS 读不到，更安全）
  res.cookies.set("blog_token", token, {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "lax",
    maxAge: 60 * 60 * 24 * 30,
    path: "/",
  });

  return res;
}
```

### 3. 创建初始用户（种子脚本）

第一次没有用户，需要创建。写一个脚本 `scripts/seed.ts`：

```ts
// scripts/seed.ts
import { PrismaClient } from "@prisma/client";
import { hash } from "bcryptjs";

const prisma = new PrismaClient();

async function main() {
  const password = await hash(process.env.BLOG_PASSWORD!, 10);
  await prisma.user.upsert({
    where: { id: "me" },
    update: {},
    create: {
      id: "me",
      name: "我",
      password,
    },
  });
  console.log("✅ 用户已创建");
}

main()
  .catch((e) => {
    console.error(e);
    process.exit(1);
  })
  .finally(() => prisma.$disconnect());
```

在 `package.json` 里加脚本：

```json
"scripts": {
  "seed": "tsx scripts/seed.ts"
}
```

安装 `tsx`（用来直接运行 TS 脚本）：

```bash
npm install -D tsx
```

运行：

```bash
npm run seed
```

## 四、路由保护（middleware）

创建 `middleware.ts`（项目根目录，和 `app` 同级）：

```ts
// middleware.ts
import { NextResponse } from "next/server";
import type { NextRequest } from "next/server";
import { jwtVerify } from "jose";

const secret = new TextEncoder().encode(process.env.JWT_SECRET!);

export async function middleware(req: NextRequest) {
  const { pathname } = req.nextUrl;

  // 登录页和 API 里的 login 不需要保护
  if (pathname === "/login" || pathname.startsWith("/api/auth/login")) {
    return NextResponse.next();
  }

  const token = req.cookies.get("blog_token")?.value;
  if (!token) {
    return NextResponse.redirect(new URL("/login", req.url));
  }

  try {
    await jwtVerify(token, secret);
    return NextResponse.next();
  } catch {
    return NextResponse.redirect(new URL("/login", req.url));
  }
}

export const config = {
  matcher: ["/((?!_next|favicon.ico|.*\\.(?:png|jpg|svg|ico)$).*)"],
};
```

> 这样任何未登录访问都会跳转到 `/login`。登录页、API 的 login 接口、静态资源除外。

## 五、登录页面

创建 `app/login/page.tsx`：

```tsx
// app/login/page.tsx
"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";

export default function LoginPage() {
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const router = useRouter();

  async function submit(e: React.FormEvent) {
    e.preventDefault();
    setError("");
    const res = await fetch("/api/auth/login", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ password }),
    });
    if (res.ok) {
      router.push("/"); // 登录成功跳首页
    } else {
      setError("密码不对，再想想");
    }
  }

  return (
    <div className="min-h-screen flex items-center justify-center p-6">
      <form onSubmit={submit} className="hand-drawn p-8 w-full max-w-sm space-y-4">
        <h1 className="font-hand text-3xl text-center">进来吧</h1>
        <input
          type="password"
          value={password}
          onChange={(e) => setPassword(e.target.value)}
          placeholder="口令"
          className="w-full bg-[var(--bg-muted)] border-2 border-[var(--ink)] rounded-lg px-3 py-2 outline-none"
        />
        {error && <p className="text-[var(--danger)] text-sm">{error}</p>}
        <button
          type="submit"
          className="w-full bg-[var(--ink)] text-[var(--bg)] border-2 border-[var(--ink)] rounded-[12px_4px_14px_6px/6px_14px_4px_12px] py-2 font-bold"
        >
          打开博客
        </button>
      </form>
    </div>
  );
}
```

## 七、关于"路线 B：Express"的对照

如果你走独立 Express，这一章的核心逻辑不变，只是：

1. Prisma schema 和 seed 完全一样（共用）。
2. 登录逻辑从 `route.ts` 搬到 `server/routes/auth.js`。
3. middleware 换成 Express 中间件（`server/middleware/auth.js`）。
4. 前端 fetch 的地址从 `/api/xxx` 改成 `http://localhost:4000/api/xxx`。

师傅建议：**先把 A 路线跑通**，理解逻辑后，B 路线就是"搬个家"。
