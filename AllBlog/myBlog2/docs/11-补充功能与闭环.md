## 一、心情日历（MoodLog）

### 1. API：`app/api/moods/route.ts`

```ts
// app/api/moods/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";
import { requireAuth } from "@/lib/auth";

// GET 某个月的心情记录
export async function GET(req: Request) {
  const { searchParams } = new URL(req.url);
  const from = searchParams.get("from");
  const to = searchParams.get("to");

  const logs = await db.moodLog.findMany({
    where: {
      date: {
        ...(from && { gte: new Date(from) }),
        ...(to && { lte: new Date(to) }),
      },
    },
  });
  return NextResponse.json({ data: logs });
}

// POST 记录/更新当天心情（一天一条，重复则覆盖）
export async function POST(req: Request) {
  if (!(await requireAuth(req))) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }
  const { mood, note } = await req.json();

  // 取当天 0 点作为 date（避免时间戳导致同一天多条）
  const today = new Date();
  today.setHours(0, 0, 0, 0);

  const log = await db.moodLog.upsert({
    where: { date: today },
    update: { mood, ...(note !== undefined && { note }) },
    create: { mood, date: today, note },
  });
  return NextResponse.json({ data: log });
}
```

> `upsert` 是关键：日期是 `@unique`，同一天重复记录会更新而不是新增。

### 2. 心情日历组件（月视图）

```tsx
// components/mood/MoodCalendar.tsx
"use client";

import { useEffect, useMemo, useState } from "react";
import { MOODS, type Mood } from "@/lib/moods";

type Log = { date: string; mood: Mood };

export default function MoodCalendar() {
  const [logs, setLogs] = useState<Log[]>([]);
  const [year, setYear] = useState(new Date().getFullYear());
  const [month, setMonth] = useState(new Date().getMonth());

  useEffect(() => {
    const from = new Date(year, month, 1).toISOString();
    const to = new Date(year, month + 1, 1).toISOString();
    fetch(`/api/moods?from=${from}&to=${to}`)
      .then((r) => r.json())
      .then((d) => setLogs(d.data));
  }, [year, month]);

  // 生成该月每天的格子
  const days = useMemo(() => {
    const first = new Date(year, month, 1);
    const count = new Date(year, month + 1, 0).getDate();
    return Array.from({ length: count }, (_, i) => {
      const d = new Date(year, month, i + 1);
      const key = d.toISOString().slice(0, 10);
      return { day: i + 1, log: logs.find((l) => l.date.slice(0, 10) === key) };
    });
  }, [logs, year, month]);

  return (
    <div className="hand-drawn p-4">
      <div className="flex items-center justify-between mb-2">
        <button onClick={() => setMonth((m) => (m === 0 ? 11 : m - 1))}>‹</button>
        <span className="font-bold">{year} 年 {month + 1} 月</span>
        <button onClick={() => setMonth((m) => (m === 11 ? 0 : m + 1))}>›</button>
      </div>
      <div className="grid grid-cols-7 gap-1 text-center text-xs">
        {["一", "二", "三", "四", "五", "六", "日"].map((w) => (
          <span key={w} className="text-[var(--ink-soft)]">{w}</span>
        ))}
        {days.map((d) => (
          <div
            key={d.day}
            className="aspect-square rounded-md flex items-center justify-center border"
            style={{
              background: d.log ? MOODS[d.log.mood].color : "transparent",
              borderColor: "var(--bg-muted)",
            }}
            title={d.log ? `${d.day} 日 · ${MOODS[d.log.mood].label}` : `${d.day} 日`}
          >
            {d.day}
          </div>
        ))}
      </div>
    </div>
  );
}
```

## 二、随机回忆 API

### `app/api/random-note/route.ts`

```ts
// app/api/random-note/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";

export async function GET() {
  // 取一条随机随笔的摘要
  const count = await db.note.count();
  if (count === 0) {
    return NextResponse.json({ data: null });
  }
  const skip = Math.floor(Math.random() * count);
  const note = await db.note.findFirst({
    where: { isDraft: false },
    skip,
    orderBy: { createdAt: "asc" },
  });
  if (!note) return NextResponse.json({ data: null });
  // 只返回一句话摘要
  const snippet = note.content.replace(/[#*`>\n]/g, "").slice(0, 60);
  return NextResponse.json({ data: snippet });
}
```

## 四、备份与导入

### 1. 导出 API：`app/api/backup/route.ts`

```ts
// app/api/backup/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";
import { requireAuth } from "@/lib/auth";

// GET 导出全部数据
export async function GET(req: Request) {
  if (!(await requireAuth(req))) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }
  const [notes, capsules, todos, moodLogs, widgets, settings] = await Promise.all([
    db.note.findMany({ include: { tags: true } }),
    db.capsule.findMany(),
    db.todo.findMany(),
    db.moodLog.findMany(),
    db.widget.findMany(),
    db.setting.findMany(),
  ]);

  const backup = {
    version: 1,
    exportedAt: new Date().toISOString(),
    data: { notes, capsules, todos, moodLogs, widgets, settings },
  };

  // 触发浏览器下载 JSON 文件
  return new NextResponse(JSON.stringify(backup, null, 2), {
    headers: {
      "Content-Type": "application/json",
      "Content-Disposition": `attachment; filename="backup-${Date.now()}.json"`,
    },
  });
}
```

### 2. 导入：前端页面上传 JSON

```tsx

async function importBackup(file: File) {
  const text = await file.text();
  const backup = JSON.parse(text);
  const res = await fetch("/api/backup", {
    method: "POST",
    headers: { "Content-Type": "application/json" },
```

导入 API（POST）：

```ts
// 追加到 backup/route.ts
export async function POST(req: Request) {
  if (!(await requireAuth(req))) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }
  const backup = await req.json();
  const { notes, capsules, todos, moodLogs, widgets } = backup.data;

  // 简单做法：逐个重建（个人博客数据量小，够用）
  await db.$transaction([
    ...notes.map((n: any) =>
      db.note.create({ data: { title: n.title, content: n.content, mood: n.mood } })
    ),
    ...capsules.map((c: any) =>
      db.capsule.create({ data: { title: c.title, content: c.content, mood: c.mood, openAt: new Date(c.openAt), isOpened: c.isOpened } })
    ),
    ...todos.map((t: any) =>
      db.todo.create({ data: { title: t.title, done: t.done } })
    ),
    ...moodLogs.map((m: any) =>
      db.moodLog.create({ data: { mood: m.mood, date: new Date(m.date) } })
    ),
    ...widgets.map((w: any) =>
      db.widget.create({ data: { type: w.type, config: w.config, position: w.position } })
    ),
  ]);

  return NextResponse.json({ data: { ok: true } });
}
```

> 导入前建议**先备份当前数据**，避免导入覆盖时丢失。师傅会在设置页加一个明确的警告提示。

## 五、主题切换（浅色/深色）

### 1. 主题切换按钮

```tsx
// components/layout/ThemeToggle.tsx
"use client";

import { useEffect, useState } from "react";

export default function ThemeToggle() {
  const [dark, setDark] = useState(false);

  useEffect(() => {
    // 从 localStorage 读上次的选择
    const saved = localStorage.getItem("theme");
    const isDark = saved === "dark";
    setDark(isDark);
    document.documentElement.classList.toggle("dark", isDark);
  }, []);

  function toggle() {
    const next = !dark;
    setDark(next);
    document.documentElement.classList.toggle("dark", next);
    localStorage.setItem("theme", next ? "dark" : "light");
  }

  return (
    <button onClick={toggle} className="border-2 border-[var(--ink)] rounded-full px-3 py-1 text-sm">
      {dark ? "☀️ 浅色" : "🌙 深色"}
    </button>
  );
}
```

> 注意：CSS 变量 `--bg` 等已经在 `globals.css` 里定义了 `.dark` 覆盖（第 05 章），这里只需切换 `html` 元素上的 `dark` class。

把 `<ThemeToggle />` 放进 `Navbar`。

## 六、设置页汇总

创建 `app/settings/page.tsx`，把备份导出、导入、主题切换放一起：

```tsx
// app/settings/page.tsx
"use client";

export default function SettingsPage() {
  function exportData() {
    window.location.href = "/api/backup"; // 触发下载
  }

  async function importData(e: React.ChangeEvent<HTMLInputElement>) {
    const file = e.target.files?.[0];
    if (!file) return;
    if (!confirm("导入会新增数据，建议先备份。确定继续？")) return;
    const text = await file.text();
    await fetch("/api/backup", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: text,
    });
    alert("导入完成");
  }

  return (
    <div className="max-w-xl mx-auto p-6 space-y-4">
      <h1 className="font-hand text-3xl">设置</h1>

      <section className="hand-drawn p-5 space-y-3">
        <h2 className="font-bold">数据备份</h2>
        <p className="text-sm text-[var(--ink-soft)]">
          你的文字是你的命，定期备份，有备无患。
        </p>
        <button onClick={exportData} className="border-2 border-[var(--ink)] rounded-lg px-4 py-2">
          ⬇ 导出全部数据
        </button>
        <label className="block">
          <span className="text-sm text-[var(--ink-soft)]">导入数据：</span>
          <input type="file" accept="application/json" onChange={importData} className="mt-2" />
        </label>
      </section>
    </div>
  );
}
```
