# 09 · 待办清单模块（手把手）

> 待办看起来最简单，但它是练"乐观更新"的最佳场所——点勾选要**立即**有反应，不能等网络。
> 这一章教会你一个前端高级但必要的技巧。

---

## 一、API 层

### `app/api/todos/route.ts`（列表 + 新建）

```ts
// app/api/todos/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";
import { requireAuth } from "@/lib/auth";

// GET 待办列表（未完成在前，按 order 排序）
export async function GET() {
  const todos = await db.todo.findMany({
    orderBy: [{ done: "asc" }, { order: "asc" }, { createdAt: "desc" }],
  });
  return NextResponse.json({ data: todos });
}

// POST 新建
export async function POST(req: Request) {
  if (!(await requireAuth(req))) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }
  const { title, dueDate } = await req.json();
  if (!title?.trim()) {
    return NextResponse.json({ error: "内容不能为空" }, { status: 400 });
  }
  const todo = await db.todo.create({
    data: { title, dueDate: dueDate ? new Date(dueDate) : null },
  });
  return NextResponse.json({ data: todo }, { status: 201 });
}
```

### `app/api/todos/[id]/route.ts`（更新 + 删除）

```ts
// app/api/todos/[id]/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";
import { requireAuth } from "@/lib/auth";

export async function PATCH(
  req: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  if (!(await requireAuth(req))) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }
  const { id } = await params;
  const body = await req.json();

  const todo = await db.todo.update({
    where: { id },
    data: {
      ...(body.done !== undefined && {
        done: body.done,
        completedAt: body.done ? new Date() : null,
      }),
      ...(body.title !== undefined && { title: body.title }),
      ...(body.dueDate !== undefined && { dueDate: body.dueDate }),
      ...(body.order !== undefined && { order: body.order }),
    },
  });
  return NextResponse.json({ data: todo });
}

export async function DELETE(
  req: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  if (!(await requireAuth(req))) {
    return NextResponse.json({ error: "未登录" }, { status: 401 });
  }
  const { id } = await params;
  await db.todo.delete({ where: { id } });
  return NextResponse.json({ data: { ok: true } });
}
```

## 二、前端：待办页面（含乐观更新）

创建 `app/todos/page.tsx`：

```tsx
// app/todos/page.tsx
"use client";

import { useEffect, useState } from "react";

type Todo = {
  id: string;
  title: string;
  done: boolean;
  dueDate: string | null;
};

export default function TodosPage() {
  const [todos, setTodos] = useState<Todo[]>([]);
  const [input, setInput] = useState("");

  async function load() {
    const res = await fetch("/api/todos");
    const d = await res.json();
    setTodos(d.data);
  }

  useEffect(() => {
    load();
  }, []);

  async function add(e: React.FormEvent) {
    e.preventDefault();
    if (!input.trim()) return;
    const res = await fetch("/api/todos", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ title: input }),
    });
    if (res.ok) {
      setInput("");
      load();
    }
  }

  // 乐观更新：先本地改，失败再回滚
  async function toggle(todo: Todo) {
    const newDone = !todo.done;
    // 1. 立即更新本地状态（用户马上看到勾选效果）
    setTodos((prev) =>
      prev.map((t) => (t.id === todo.id ? { ...t, done: newDone } : t))
    );

    // 2. 同步到后端
    const res = await fetch(`/api/todos/${todo.id}`, {
      method: "PATCH",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ done: newDone }),
    });

    // 3. 失败则回滚
    if (!res.ok) {
      setTodos((prev) =>
        prev.map((t) => (t.id === todo.id ? { ...t, done: !newDone } : t))
      );
    }
  }

  async function remove(id: string) {
    await fetch(`/api/todos/${id}`, { method: "DELETE" });
    setTodos((prev) => prev.filter((t) => t.id !== id));
  }

  return (
    <div className="max-w-xl mx-auto p-6 space-y-4">
      <h1 className="font-hand text-3xl">待办清单</h1>

      <form onSubmit={add} className="flex gap-2">
        <input
          value={input}
          onChange={(e) => setInput(e.target.value)}
          placeholder="要做的事……"
          className="flex-1 bg-[var(--bg-muted)] border-2 border-[var(--ink)] rounded-lg px-3 py-2 outline-none"
        />
        <button
          type="submit"
          className="bg-[var(--ink)] text-[var(--bg)] border-2 border-[var(--ink)] rounded-lg px-4"
        >
          添加
        </button>
      </form>

      <ul className="space-y-2">
        {todos.map((todo) => (
          <li
            key={todo.id}
            className="hand-drawn p-3 flex items-center gap-3"
          >
            {/* 手绘勾选框 */}
            <button
              onClick={() => toggle(todo)}
              className="w-6 h-6 border-2 border-[var(--ink)] rounded-md flex items-center justify-center shrink-0"
              aria-label={todo.done ? "取消完成" : "标记完成"}
            >
              {todo.done && (
                <svg width="16" height="16" viewBox="0 0 16 16">
                  <path
                    d="M3 8 L6.5 11.5 L13 4"
                    fill="none"
                    stroke="var(--ink)"
                    strokeWidth="2.5"
                    strokeLinecap="round"
                    strokeLinejoin="round"
                  />
                </svg>
              )}
            </button>

            <span
              className={`flex-1 ${
                todo.done ? "line-through text-[var(--ink-soft)]" : ""
              }`}
            >
              {todo.title}
            </span>

            <button
              onClick={() => remove(todo.id)}
              className="text-[var(--danger)] text-sm opacity-60 hover:opacity-100"
            >
              删除
            </button>
          </li>
        ))}
      </ul>

      {todos.length === 0 && (
        <p className="text-[var(--ink-soft)]">清单空空的，很清静。</p>
      )}
    </div>
  );
}
```

## 三、乐观更新详解（师傅划重点）

什么是"乐观更新"？一句话：**先假装成功，失败再改回来。**

```
普通做法（悲观）：点勾选 → 等后端返回 → 才更新界面（会卡顿）
乐观做法：       点勾选 → 立即更新界面 → 后台悄悄同步 → 失败才回滚
```

对比一下体感：
- 悲观：点一下，转圈 300ms，勾才出现 → 觉得"卡"。
- 乐观：点一下，勾立刻出现 → 觉得"顺滑"。

**但代价是**：多写了"失败回滚"的逻辑。这个 trade-off 很值，因为它决定了手感的差距。

## 四、进阶：勾选动画

如果你想让打勾有"手绘"动画，给 SVG 的 `path` 加描边动画：

```css
/* 勾线从无到有画出来 */
.check-path {
  stroke-dasharray: 20;
  stroke-dashoffset: 20;
  animation: draw 0.3s ease-out forwards;
}
@keyframes draw {
  to { stroke-dashoffset: 0; }
}
```

然后给 `<path>` 加 `className="check-path"`。
