## 一、API 层（增删改查四件套）

### 1. 列表 + 新建：`app/api/notes/route.ts`

```ts
// app/api/notes/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";

// GET 获取随笔列表（支持按标签、搜索筛选）
export async function GET(req: Request) {
  const { searchParams } = new URL(req.url);
  const tag = searchParams.get("tag");
  const q = searchParams.get("q");

  const notes = await db.note.findMany({
    where: {
      // 有标签就按标签筛
      ...(tag ? { tags: { some: { name: tag } } } : {}),
      // 有搜索词就按标题/正文模糊搜
      ...(q
        ? {
            OR: [
              { title: { contains: q } },
              { content: { contains: q } },
            ],
          }
        : {}),
    },
    include: { tags: true },
    orderBy: { createdAt: "desc" },
  });

  return NextResponse.json({ data: notes });
}

// POST 新建随笔
export async function POST(req: Request) {
  const { title, content, mood = "CALM", tags = [] } = await req.json();

  if (!title?.trim() || !content?.trim()) {
    return NextResponse.json(
      { error: "标题和正文都不能为空" },
      { status: 400 }
    );
  }

  const note = await db.note.create({
    data: {
      title,
      content,
      mood,
      tags: {
        // 标签：已存在就连接，不存在就新建（connectOrCreate）
        connectOrCreate: tags.map((name: string) => ({
          where: { name },
          create: { name },
        })),
      },
    },
    include: { tags: true },
  });

  return NextResponse.json({ data: note }, { status: 201 });
}
```

> 重点理解 `connectOrCreate`：这是 Prisma 处理"多对多 + 自动去重"的标准写法，标签不存在就自动建，存在就复用。

### 2. 单个（详情/更新/删除）：`app/api/notes/[id]/route.ts`

```ts
// app/api/notes/[id]/route.ts
import { NextResponse } from "next/server";
import { db } from "@/lib/db";

// GET 详情
export async function GET(
  _req: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params;
  const note = await db.note.findUnique({
    where: { id },
    include: { tags: true },
  });
  if (!note) {
    return NextResponse.json({ error: "没找到这篇随笔" }, { status: 404 });
  }
  return NextResponse.json({ data: note });
}

// PATCH 更新（部分字段）
export async function PATCH(
  req: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params;
  const body = await req.json();

  const note = await db.note.update({
    where: { id },
    data: {
      ...(body.title !== undefined && { title: body.title }),
      ...(body.content !== undefined && { content: body.content }),
      ...(body.mood !== undefined && { mood: body.mood }),
    },
    include: { tags: true },
  });

  return NextResponse.json({ data: note });
}

// DELETE 删除
export async function DELETE(
  _req: Request,
  { params }: { params: Promise<{ id: string }> }
) {
  const { id } = await params;
  await db.note.delete({ where: { id } });
  return NextResponse.json({ data: { ok: true } });
}
```

> 注意：Next 15 里 `params` 是 Promise，要用 `await params` 取值。这是新手最常踩的坑。

## 二、前端：随笔列表页

创建 `app/notes/page.tsx`：

```tsx
// app/notes/page.tsx
"use client";

import { useEffect, useState } from "react";
import Link from "next/link";
import MoodIcon from "@/components/mood/MoodIcon";
import type { Mood } from "@/lib/moods";

type Note = {
  id: string;
  title: string;
  content: string;
  mood: Mood;
  createdAt: string;
  tags: { id: string; name: string }[];
};

export default function NotesPage() {
  const [notes, setNotes] = useState<Note[]>([]);
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    fetch("/api/notes")
      .then((r) => r.json())
      .then((d) => {
        setNotes(d.data);
        setLoading(false);
      });
  }, []);

  if (loading) return <p className="p-8">加载中……</p>;

  return (
    <div className="max-w-2xl mx-auto p-6 space-y-4">
      <div className="flex items-center justify-between">
        <h1 className="font-hand text-3xl">随笔</h1>
        <Link
          href="/notes/new"
          className="bg-[var(--ink)] text-[var(--bg)] border-2 border-[var(--ink)] rounded-[12px_4px_14px_6px/6px_14px_4px_12px] px-4 py-2 shadow-[3px_3px_0_var(--ink)]"
        >
          + 写一篇
        </Link>
      </div>

      {notes.length === 0 && (
        <p className="text-[var(--ink-soft)]">还没有随笔，写第一篇吧。</p>
      )}

      {notes.map((n) => (
        <Link key={n.id} href={`/notes/${n.id}`} className="block">
          <article className="hand-drawn hand-drawn-hover p-5">
            <div className="flex items-center gap-3">
              <MoodIcon mood={n.mood} size={28} />
              <h2 className="font-bold text-lg">{n.title}</h2>
            </div>
            <p className="text-[var(--ink-soft)] text-sm mt-2 line-clamp-2">
              {n.content.replace(/[#*`>]/g, "").slice(0, 80)}
            </p>
            <div className="flex items-center gap-2 mt-3 text-xs text-[var(--ink-soft)]">
              <span>{new Date(n.createdAt).toLocaleDateString("zh-CN")}</span>
              {n.tags.map((t) => (
                <span key={t.id} className="px-2 py-0.5 bg-[var(--bg-muted)] rounded-full">
                  #{t.name}
                </span>
              ))}
            </div>
          </article>
        </Link>
      ))}
    </div>
  );
}
```

## 三、前端：写作页（新建 + 编辑二合一）

创建 `app/notes/new/page.tsx`：

```tsx
// app/notes/new/page.tsx
"use client";

import { useState } from "react";
import { useRouter } from "next/navigation";
import { MOOD_LIST } from "@/lib/moods";
import MoodIcon from "@/components/mood/MoodIcon";

export default function NewNotePage() {
  const [title, setTitle] = useState("");
  const [content, setContent] = useState("");
  const [mood, setMood] = useState("CALM");
  const [tags, setTags] = useState("");
  const [saving, setSaving] = useState(false);
  const router = useRouter();

  async function save(draft = false) {
    if (!title.trim() || !content.trim()) {
      alert("标题和正文不能为空");
      return;
    }
    setSaving(true);
    const res = await fetch("/api/notes", {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({
        title,
        content,
        mood,
        isDraft: draft,
        tags: tags.split(/[,，\s]+/).filter(Boolean),
      }),
    });
    setSaving(false);
    if (res.ok) {
      router.push("/notes"); // 保存成功回列表
      router.refresh();
    } else {
      alert("保存失败");
    }
  }

  return (
    <div className="max-w-2xl mx-auto p-6 space-y-4">
      <h1 className="font-hand text-3xl">写点什么</h1>

      {/* 心情选择 */}
      <div className="flex gap-2">
        {MOOD_LIST.map((m) => (
          <button
            key={m.key}
            onClick={() => setMood(m.key)}
            className={`p-2 rounded-full border-2 ${
              mood === m.key ? "border-[var(--ink)]" : "border-transparent"
            }`}
            title={m.label}
          >
            <MoodIcon mood={m.key} size={28} />
          </button>
        ))}
      </div>

      <input
        value={title}
        onChange={(e) => setTitle(e.target.value)}
        placeholder="标题"
        className="w-full text-2xl font-bold bg-transparent outline-none"
      />

      <input
        value={tags}
        onChange={(e) => setTags(e.target.value)}
        placeholder="标签，用逗号分隔，如：生活, 编程"
        className="w-full bg-[var(--bg-muted)] border-2 border-[var(--ink)] rounded-lg px-3 py-2"
      />

      <textarea
        value={content}
        onChange={(e) => setContent(e.target.value)}
        placeholder="正文……（支持 Markdown）"
        rows={15}
        className="w-full paper p-4 outline-none resize-y leading-relaxed"
      />

      <div className="flex gap-3">
        <button
          onClick={() => save(false)}
          disabled={saving}
          className="bg-[var(--ink)] text-[var(--bg)] border-2 border-[var(--ink)] rounded-[12px_4px_14px_6px/6px_14px_4px_12px] px-5 py-2"
        >
          {saving ? "保存中…" : "保存"}
        </button>
        <button
          onClick={() => save(true)}
          disabled={saving}
          className="border-2 border-[var(--ink)] rounded-[12px_4px_14px_6px/6px_14px_4px_12px] px-5 py-2"
        >
          存草稿
        </button>
      </div>
    </div>
  );
}
```

## 四、前端：详情页（渲染 Markdown + 编辑/删除）

创建 `app/notes/[id]/page.tsx`：

```tsx
// app/notes/[id]/page.tsx
"use client";

import { useEffect, useState } from "react";
import { useParams, useRouter } from "next/navigation";
import ReactMarkdown from "react-markdown";
import remarkGfm from "remark-gfm";
import MoodIcon from "@/components/mood/MoodIcon";
import type { Mood } from "@/lib/moods";

type Note = {
  id: string;
  title: string;
  content: string;
  mood: Mood;
  createdAt: string;
  tags: { id: string; name: string }[];
};

export default function NoteDetailPage() {
  const { id } = useParams<{ id: string }>();
  const [note, setNote] = useState<Note | null>(null);
  const router = useRouter();

  useEffect(() => {
    fetch(`/api/notes/${id}`)
      .then((r) => r.json())
      .then((d) => setNote(d.data));
  }, [id]);

  async function remove() {
    if (!confirm("确定删除这篇随笔吗？删了就没了。")) return;
    await fetch(`/api/notes/${id}`, { method: "DELETE" });
    router.push("/notes");
  }

  if (!note) return <p className="p-8">加载中……</p>;

  return (
    <div className="max-w-2xl mx-auto p-6">
      <article className="paper p-8">
        <div className="flex items-center justify-between">
          <div className="flex items-center gap-3">
            <MoodIcon mood={note.mood} size={32} />
            <h1 className="font-hand text-3xl">{note.title}</h1>
          </div>
          <button onClick={remove} className="text-[var(--danger)] text-sm">
            删除
          </button>
        </div>
        <p className="text-xs text-[var(--ink-soft)] mt-1">
          {new Date(note.createdAt).toLocaleString("zh-CN")}
        </p>
        <div className="prose prose-slate max-w-none mt-6 leading-relaxed">
          <ReactMarkdown remarkPlugins={[remarkGfm]}>{note.content}</ReactMarkdown>
        </div>
        {note.tags.length > 0 && (
          <div className="flex gap-2 mt-6">
            {note.tags.map((t) => (
              <span key={t.id} className="px-2 py-0.5 bg-[var(--bg-muted)] rounded-full text-sm">
                #{t.name}
              </span>
            ))}
          </div>
        )}
      </article>
    </div>
  );
}
```

> 需要 Markdown 样式（`prose`），安装 typography 插件：
> ```bash
> npm install -D @tailwindcss/typography
> ```
> 并在 `globals.css` 里加 `@plugin "@tailwindcss/typography";`。

## 五、给随笔的 API 加上"登录校验"

目前 API 还没做鉴权（middleware 只保护页面）。为安全，在 API 里也校验。创建一个辅助函数：

```ts
// lib/auth.ts
import { jwtVerify } from "jose";

const secret = new TextEncoder().encode(process.env.JWT_SECRET!);

export async function requireAuth(req: Request) {
  const token = req.headers.get("cookie")?.match(/blog_token=([^;]+)/)?.[1];
  if (!token) return false;
  try {
    await jwtVerify(token, secret);
    return true;
  } catch {
    return false;
  }
}
```

在 POST/PATCH/DELETE 里加：

```ts
// 每个写操作 API 的开头
if (!(await requireAuth(req))) {
  return NextResponse.json({ error: "未登录" }, { status: 401 });
}
```

> GET 列表可以不加（自己的博客），但写操作必须加，防止别人用工具直接调用 API 改你数据。
