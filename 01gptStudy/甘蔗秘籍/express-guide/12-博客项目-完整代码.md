# 12 小型博客项目：可运行后端

本节给出“最小但完整”的实现。为了让初学者先掌握 Express，数据先放在内存数组；下一节再替换成数据库。重启服务会丢数据，这是刻意的教学限制。

## 1. 创建项目

```bash
mkdir blog-api && cd blog-api
npm init -y
npm install express bcrypt jsonwebtoken
npm install --save-dev supertest
```

## 2. `src/app.js`

```js
const express = require('express');
const bcrypt = require('bcrypt');
const jwt = require('jsonwebtoken');

const app = express();
app.use(express.json({ limit: '1mb' }));

const users = [];
const posts = [];
let nextUserId = 1;
let nextPostId = 1;
const jwtSecret = process.env.JWT_SECRET || 'dev-only-change-me';

class AppError extends Error {
  constructor(status, code, message) { super(message); this.status = status; this.code = code; }
}

function asyncRoute(handler) {
  return (req, res, next) => Promise.resolve(handler(req, res, next)).catch(next);
}

function validateCredentials(body) {
  const username = typeof body.username === 'string' ? body.username.trim() : '';
  const password = typeof body.password === 'string' ? body.password : '';
  if (!/^[a-zA-Z0-9_]{3,30}$/.test(username)) throw new AppError(400, 'INVALID_USERNAME', '用户名需为 3-30 位字母、数字或下划线');
  if (password.length < 8 || password.length > 72) throw new AppError(400, 'INVALID_PASSWORD', '密码需为 8-72 位');
  return { username, password };
}

function requireAuth(req, res, next) {
  const header = req.get('authorization') || '';
  const token = header.startsWith('Bearer ') ? header.slice(7) : '';
  if (!token) return next(new AppError(401, 'UNAUTHORIZED', '请先登录'));
  try { req.user = jwt.verify(token, jwtSecret); next(); }
  catch { next(new AppError(401, 'UNAUTHORIZED', '登录已失效')); }
}

app.get('/health', (req, res) => res.json({ ok: true }));

app.post('/api/auth/register', asyncRoute(async (req, res) => {
  const { username, password } = validateCredentials(req.body || {});
  if (users.some(u => u.username === username)) throw new AppError(409, 'USERNAME_EXISTS', '用户名已存在');
  const user = { id: nextUserId++, username, passwordHash: await bcrypt.hash(password, 10) };
  users.push(user);
  res.status(201).json({ data: { id: user.id, username: user.username } });
}));

app.post('/api/auth/login', asyncRoute(async (req, res) => {
  const { username, password } = validateCredentials(req.body || {});
  const user = users.find(u => u.username === username);
  if (!user || !(await bcrypt.compare(password, user.passwordHash))) throw new AppError(401, 'INVALID_LOGIN', '用户名或密码错误');
  const token = jwt.sign({ sub: user.id, username: user.username }, jwtSecret, { expiresIn: '2h' });
  res.json({ data: { token } });
}));

function publicPost(post) { return { id: post.id, authorId: post.authorId, title: post.title, content: post.content, published: post.published, createdAt: post.createdAt, updatedAt: post.updatedAt }; }

app.get('/api/posts', (req, res) => {
  const page = Math.max(1, Number.parseInt(req.query.page || '1', 10));
  const pageSize = Math.min(50, Math.max(1, Number.parseInt(req.query.pageSize || '10', 10)));
  const data = posts.filter(p => p.published).slice((page - 1) * pageSize, page * pageSize).map(publicPost);
  res.json({ data, meta: { page, pageSize, total: posts.filter(p => p.published).length } });
});

app.get('/api/posts/:id', (req, res, next) => {
  const post = posts.find(p => p.id === Number(req.params.id));
  if (!post || !post.published) return next(new AppError(404, 'NOT_FOUND', '文章不存在'));
  res.json({ data: publicPost(post) });
});

app.post('/api/posts', requireAuth, (req, res, next) => {
  const title = typeof req.body?.title === 'string' ? req.body.title.trim() : '';
  const content = typeof req.body?.content === 'string' ? req.body.content.trim() : '';
  if (!title || title.length > 120 || !content) return next(new AppError(400, 'VALIDATION_ERROR', '标题和正文不能为空，标题不超过 120 字'));
  const now = new Date().toISOString();
  const post = { id: nextPostId++, authorId: Number(req.user.sub), title, content, published: false, createdAt: now, updatedAt: now };
  posts.push(post);
  res.status(201).json({ data: publicPost(post) });
});

function findOwnedPost(req, next) {
  const post = posts.find(p => p.id === Number(req.params.id));
  if (!post) return next(new AppError(404, 'NOT_FOUND', '文章不存在'));
  if (post.authorId !== Number(req.user.sub)) return next(new AppError(403, 'FORBIDDEN', '只能操作自己的文章'));
  req.post = post; next();
}

app.patch('/api/posts/:id', requireAuth, findOwnedPost, (req, res, next) => {
  const title = typeof req.body?.title === 'string' ? req.body.title.trim() : req.post.title;
  const content = typeof req.body?.content === 'string' ? req.body.content.trim() : req.post.content;
  if (!title || title.length > 120 || !content) return next(new AppError(400, 'VALIDATION_ERROR', '标题和正文不能为空'));
  req.post.title = title; req.post.content = content; req.post.updatedAt = new Date().toISOString();
  res.json({ data: publicPost(req.post) });
});

app.delete('/api/posts/:id', requireAuth, findOwnedPost, (req, res) => {
  posts.splice(posts.indexOf(req.post), 1); res.status(204).send();
});

app.post('/api/posts/:id/publish', requireAuth, findOwnedPost, (req, res) => {
  req.post.published = true; req.post.updatedAt = new Date().toISOString();
  res.json({ data: publicPost(req.post) });
});

app.use((req, res) => res.status(404).json({ error: { code: 'NOT_FOUND', message: '接口不存在' } }));
app.use((err, req, res, next) => {
  if (res.headersSent) return next(err);
  const status = Number.isInteger(err.status) ? err.status : 500;
  console.error(err);
  res.status(status).json({ error: { code: err.code || 'INTERNAL_ERROR', message: status === 500 ? '服务器内部错误' : err.message } });
});

module.exports = app;
```

## 3. `src/server.js`

```js
const app = require('./app');
const port = Number(process.env.PORT || 3000);
app.listen(port, () => console.log(`Blog API listening on http://localhost:${port}`));
```

启动：

```bash
JWT_SECRET="replace-with-a-long-random-secret" node src/server.js
```

Windows PowerShell：

```powershell
$env:JWT_SECRET = "replace-with-a-long-random-secret"; node src/server.js
```

## 4. 手工验证顺序

```text
GET /health
POST /api/auth/register
POST /api/auth/login  -> 复制 token
POST /api/posts       -> 带 Authorization: Bearer token
POST /api/posts/:id/publish
GET /api/posts
```
